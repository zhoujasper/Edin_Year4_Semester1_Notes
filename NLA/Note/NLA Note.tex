\section{Background}

\begin{MyTh}{Floating Point ($\F$)}
    A number = mantissa $\times$ base$^{\text{exponent}}$ \quad \comment{(e.g. $2048=1\times 2^{11}$ \ps{mantissa=1, base=2, exponent=11})}
\end{MyTh}

\begin{MyTh}{Assumptions}
    \comment{There is $\varepsilon_m>0$ (machine epsilon)} \quad \ok{A1}: $\forall\alpha\in\R$, $\exists \varepsilon\in(-\varepsilon_m,\varepsilon_m)$ s.t. $fl:\R\to\F$ by $fl(\alpha)=\alpha(1+\varepsilon)$
    
    \hspace{7.4cm} \ok{A2}: $\forall\alpha,\beta\in\F$, $\forall *\in\{+,-,\times,\div\}$, $\exists \varepsilon\in(-\varepsilon_m,\varepsilon_m)$ s.t. $\alpha \circledast \beta= (\alpha* \beta)(1+\varepsilon)$

    \vspace{-0.1cm}\hspace{2.38cm}\ps{A1: The rounding error in $\alpha$ is relative to the size of $\alpha$ ($|\alpha-fl(\alpha)|<|\alpha|\varepsilon_m$) \quad A2: The error in $\alpha \circledast \beta$ is relative to the size of $\alpha * \beta$ ($|\alpha \circledast \beta - \alpha * \beta|\le|\alpha * \beta|\varepsilon_m$)}
\end{MyTh}

\begin{MyTh}{IEEE 754}
    \ok{64-bit}: denote $\F$. \quad \comment{1-bit}: sign \quad \comment{52-bit}: mantissa \quad \comment{11-bit}: exponent \ps{(base 2)}. \quad \small{$\varepsilon_m\approx 2.22\times 10^{-16}$}
\end{MyTh}

\begin{MyTh}{Cost of Algorithm ($C$)}
    $C:=$ number of floating point operations (FLOPs) \qquad If $C(n)\le\alpha f(n)$, $\Rightarrow$ $C(n)=\Ocal(f(n))$. \ps{n is problem size}
\end{MyTh}

\begin{MyTh}{Norms}
    $\|\mathbf{x}\|_p:=(\sum_{i=1}^n|x_i|^p)^{1/p}$, $p\ge1$ \quad $\|\mathbf{x}\|_\infty:=\max_{1\le i\le n}|x_i|$ \quad $\|\mathbf{x}\|_2:=(\sum_{i=1}^n|x_i|^2)^{1/2}$ \quad $\|\mathbf{x}\|_1:=\sum_{i=1}^n|x_i|$

    \vspace{-10pt}
    \begin{minipage}[t]{0.4\textwidth}
        \begin{enumerate}
            \item $\|\mathbf{x}\|_p\ge0$ \quad $\|\mathbf{x}\|_p=0 \Leftrightarrow \mathbf{x}=\mathbf{0}$
            \item $\|\alpha\mathbf{x}\|_p=|\alpha|\|\mathbf{x}\|_p$ for $\alpha\in\R$
            \item $\|\mathbf{x}+\mathbf{y}\|_p\le\|\mathbf{x}\|_p+\|\mathbf{y}\|_p$ \quad \ps{(Triangle inequality)}
            \item \ok{Induced Norm}: $\|\mathbf{A}\|_p:=\max\limits_{\mathbf{x}\ne0}\frac{\|\mathbf{A}\mathbf{x}\|_p}{\|\mathbf{x}\|_p}$
        \end{enumerate}    
    \end{minipage}
    \hfill
    \begin{minipage}[t]{0.6\textwidth}
        \begin{enumerate}
            \item $\|\mathbf{A}_{m\times n}\|_\infty:=\max\limits_{1\le j\le m}\sum_{i=1}^n|a_{ij}|$ \quad \comment{(maximum row sum)}
            \item $\|\mathbf{A}_{m\times n}\|_1:=\max\limits_{1\le i\le n}\sum_{j=1}^m|a_{ij}|$ \quad \comment{(maximum column sum)}
            \item $\|\mathbf{A}_{m\times n}\|_2:=\sqrt{\rho(\mathbf{A}^\top\mathbf{A})}$ \quad \comment{$\rho(\mathbf{A}^\top\mathbf{A})=\max\{|\lambda|:\lambda \text{ is eigenvalue of } \mathbf{A}^\top\mathbf{A}\}$}
        \end{enumerate}
    \end{minipage}
    
\end{MyTh}


\section{Methods for Systems of Linear Equations (SLE)}

\qquad \ \ \ok{Ideal:} We want to solve SLE: $\mathbf{A}\mathbf{x}=\mathbf{b}$, where $\mathbf{A}_{n\times n}$ is invertible, $\mathbf{x},\mathbf{b}\in\R^n$
\vspace{-0.15cm}

\subsection{LU Factorization | Gaussian Elimination}

\begin{MyTh}{Def of LU}
    If $\mathbf{A}_{n\times n}$ is invertible, then $^1\exists \ \mathbf{L}$: \comment{单位下三角}, $^2\exists \ \mathbf{U}$: \comment{可逆上三角}. \ s.t. \ $\mathbf{A}=\mathbf{L}\mathbf{U}$ \ok{(unique)}
\end{MyTh}

\begin{MyTh}{LU Factorization}
    By Gaussian elimination, $\mathbf{A}_{n\times n}=(\prod_{k=1}^{n-1}\mathbf{L}_k)^{-1}\mathbf{U}=:\mathbf{L}\mathbf{U}$ \ \ \ps{$\mathbf{L}_k$: Lower triangular with 1s on the diagonal (by row operation on single column)}
    
    \begin{enumerate}
        \item 如果$\mathbf{L}$是\comment{单位下三角}矩阵,且只有第$k$列有非零元素在对角线下,则$\mathbf{L}^{-1}$矩阵是$\mathbf{L}$对角线不动,其余为相反数.
        \item 如果$\mathbf{L}_1,\mathbf{L}_2$矩阵是\comment{单位下三角}矩阵,且$\mathbf{L}_1|\mathbf{L}_2$在第$1\sim k|k+1\sim n$列有非零元素在对角线下,则$\mathbf{L}_1\mathbf{L}_2$矩阵是他们的"拼接".
        \item By LU factorization, $\mathbf{A}=\mathbf{L}\mathbf{U}$ \ \ $\Rightarrow$ \ \ By solving \ok{$\mathbf{L}\mathbf{y}=\mathbf{b}$} and \ok{$\mathbf{U}\mathbf{x}=\mathbf{y}$}, we can solve SLE \ok{$\mathbf{A}\mathbf{x}=\mathbf{b}$}. \quad \ps{用$\mathbf{FS}$和$\mathbf{BS}$算法}
    \end{enumerate}

    \vspace{-6pt}
    \begin{minipage}[t]{0.4\textwidth}
    \[e.g. \ (1) \quad
    \begin{bmatrix}
    1 & 0 & 0 \\
    2 & 1 & 0 \\
    3 & 0 & 1
    \end{bmatrix}^{-1}
    =
    \begin{bmatrix}
    1 & 0 & 0 \\
    -2 & 1 & 0 \\
    -3 & 0 & 1
    \end{bmatrix}
    \]
    \end{minipage}
    \hfill
    \begin{minipage}[t]{0.6\textwidth}
    \[(2) \quad
    \begin{bmatrix}
    1 & 0 & 0 \\
    2 & 1 & 0 \\
    4 & 0 & 1
    \end{bmatrix}
    \begin{bmatrix}
    1 & 0 & 0 \\
    0 & 1 & 0 \\
    0 & 5 & 1
    \end{bmatrix}
    =
    \begin{bmatrix}
    1 & 0 & 0 \\
    2 & 1 & 0 \\
    4 & 5 & 1
    \end{bmatrix}
    \]
    \end{minipage}

\end{MyTh}

\begin{MyTh}{Error Analysis of GE}
  Gaussian elimination is an \ok{unstable} algorithm. \ps{(i.e. floating point arithmetic can have a disastrous effect on the computed solution.)}
\end{MyTh}




\section{Algorithms}

\begin{multicols}{3}

% ----- Algorithm DP -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{DP} Dot product \quad \ps{$C = \sum_{i=1}^{n} 2 = 2n=\Ocal(n)$}}
\textbf{Input:} $\mathbf{x}, \mathbf{y} \in \mathbb{R}^n$ \\
\textbf{Output:} $s = \mathbf{x}\cdot\mathbf{y} = \mathbf{x}^\top \mathbf{y}$
\begin{algorithmic}[1]
  \State $s = 0$
  \For{$i = 1, \dots, n$}
    \State $s = s + x_i y_i$
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm MV -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{MV} Matrix-vector multiplication \vspace{-0.1cm} \\ \ps{$C = \sum_{i=1}^{m} 2n = 2nm=\Ocal(mn)$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{m \times n}$, $\mathbf{x} \in \mathbb{R}^n$ \\
\textbf{Output:} $\mathbf{y} = \mathbf{A}\mathbf{x}$
\begin{algorithmic}[1]
  \For{$i = 1, \dots, m$}
    \State $y_i = \mathbf{a}_i^\top \mathbf{x}$ \codealg{DP}
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm MM -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{MM} Matrix-matrix multiplication \vspace{-0.1cm} \\ \ps{$C = \sum_{i=1}^{m} \sum_{j=1}^{p} 2n = 2mnp=\Ocal(mnp)$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{m \times n}$, $\mathbf{B} \in \mathbb{R}^{n \times p}$ \\
\textbf{Output:} $\mathbf{C} = \mathbf{A} \mathbf{B}$
\begin{algorithmic}[1]
  \For{$i = 1, \dots, m$}
    \For{$j = 1, \dots, p$}
      \State $c_{ij} = \mathbf{a}_i^\top \mathbf{b}_j$ \codealg{DP}
    \EndFor
  \EndFor
\end{algorithmic}
\end{algorithm}

\end{multicols}

\begin{multicols}{3}

% ----- Algorithm LU -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{LU} LU factorisation \vspace{-0.1cm} \\ \ps{$C(n) = \sum^{n-1}_{k=1}(n-k)(1+2(n-k+1)) = \frac{2}{3}n^3+\frac{1}{2}n^2-\frac{7}{6}n$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{n \times n}$ \\
\textbf{Output:} $\mathbf{L}, \mathbf{U} \in \mathbb{R}^{n \times n}$ \ \ s.t. \ \ $\mathbf{A} = \mathbf{LU}$
\begin{algorithmic}[1]
  \State $\mathbf{L} = \mathbf{I},\ \mathbf{U} = \mathbf{A}$
  \For{$k = 1, \dots, n - 1$}
    \For{$j = k + 1, \dots, n$}
      \State $l_{jk} = u_{jk} / u_{kk}$
      \State $(u_{jk}, \dots, u_{jn}) = (u_{jk}, \dots, u_{jn}) - l_{jk}(u_{kk}, \dots, u_{kn})$
    \EndFor
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm FS -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{FS} Forward substitution \vspace{-0.1cm} \\ \ps{$C(n) = \sum_{j=1}^{n}[2(j-1)+1]=n^2=\Ocal(n^2)$}}
\textbf{Input:} $\mathbf{L} \in \mathbb{R}^{n \times n}$ (lower triangular), $\mathbf{b} \in \mathbb{R}^n$ \\
\textbf{Output:} $\mathbf{y} \in \mathbb{R}^n$ such that $\mathbf{Ly} = \mathbf{b}$
\begin{algorithmic}[1]
  \For{$j = 1, \dots, n$}
    \State $y_j = \frac{1}{l_{jj}} \left( b_j - \sum_{k = 1}^{j - 1} l_{jk} y_k \right)$
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm BS -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{BS} Back substitution \vspace{-0.1cm} \\ \ps{$C(n) = \sum_{j=1}^n[2(n-j)+1]=n^2=\Ocal(n^2)$}}
\textbf{Input:} $\mathbf{U} \in \mathbb{R}^{n \times n}$ (upper triangular), $\mathbf{y} \in \mathbb{R}^n$ \\
\textbf{Output:} $\mathbf{x} \in \mathbb{R}^n$ such that $\mathbf{Ux} = \mathbf{y}$
\begin{algorithmic}[1]
  \For{$j = n, \dots, 1$}
    \State $x_j = \frac{1}{u_{jj}} \left( y_j - \sum_{k = j + 1}^{n} u_{jk} x_k \right)$
  \EndFor
\end{algorithmic}
\end{algorithm}

\end{multicols}

\vspace{-0.8cm}
\begin{multicols}{3}

% ----- Algorithm GE -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{GE} Gaussian elimination \vspace{-0.1cm} \\ \ps{$C(n) = \frac{2}{3}n^3 + \frac{5}{2}n^2 - \frac{7}{6}n=\Ocal(n^3)$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{n \times n},\ \mathbf{b} \in \mathbb{R}^n$ \\
\textbf{Output:} $\mathbf{x} \in \mathbb{R}^n$ \ \ s.t. \ \ $\mathbf{Ax} = \mathbf{b}$
\begin{algorithmic}[1]
  \State Find LU factorisation of $\mathbf{A}$. \codealg{LU}
  \State Solve $\mathbf{Ly} = \mathbf{b}$ using forward substitution. \codealg{FS}
  \State Solve $\mathbf{Ux} = \mathbf{y}$ using back substitution. \codealg{BS}
\end{algorithmic}
\end{algorithm}

\end{multicols}
