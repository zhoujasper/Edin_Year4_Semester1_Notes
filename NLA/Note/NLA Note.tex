\section{Background}

\begin{MyTh}{Floating Point ($\F$)}
    A number = mantissa $\times$ base$^{\text{exponent}}$ \quad \comment{(e.g. $2048=1\times 2^{11}$ \ps{mantissa=1, base=2, exponent=11})}
\end{MyTh}

\begin{MyTh}{Assumptions}
    \comment{There is $\varepsilon_m>0$ \ok{(machine epsilon)}} \ \ \ok{A1}: $\forall\alpha\in\R$, $\exists \varepsilon\in(-\varepsilon_m,\varepsilon_m)$ s.t. $fl:\R\to\F$ by $fl(\alpha)=\alpha(1+\varepsilon)$
    
    \qquad\qquad\qquad\quad {\small $\varepsilon_m\approx 2.22\times 10^{-16}$} \hspace{65pt} \ok{A2}: $\forall\alpha,\beta\in\F$, $\forall *\in\{+,-,\times,\div\}$, $\exists \varepsilon\in(-\varepsilon_m,\varepsilon_m)$ s.t. $\alpha \circledast \beta= (\alpha* \beta)(1+\varepsilon)$

    \vspace{-0.09cm}\hspace{2cm}\ps{A1: The rounding error in $\alpha$ is relative to the size of $\alpha$ ($|\alpha-fl(\alpha)|<|\alpha|\varepsilon_m$) \quad A2: The error in $\alpha \circledast \beta$ is relative to the size of $\alpha * \beta$ ($|\alpha \circledast \beta - \alpha * \beta|\le|\alpha * \beta|\varepsilon_m$)}
\end{MyTh}

\begin{MyTh}{Cost of Algorithm ($C$)}
    $C:=$ number of floating point operations (FLOPs) \qquad If $C(n)\le\alpha f(n)$, $\Rightarrow$ $C(n)=\Ocal(f(n))$.

    \quad $\cdot$ We consider that: \ok{交换/赋值/相反数} no cost \ps{(i.e. Permutation $\mathbf{P}$ is free)}, \ \ \ok{比较两数} 1 FLOPs, \ \ \ok{$\sqrt{~~~~}$} 1 FLOPs.
\end{MyTh}

\begin{MyTh}{Norms}
    $\|\mathbf{x}\|_p:=(\sum_{i=1}^n|x_i|^p)^{1/p}$, $p\ge1$ \quad $\|\mathbf{x}\|_\infty:=\max_{1\le i\le n}|x_i|$ \quad $\|\mathbf{x}\|_2:=(\sum_{i=1}^n|x_i|^2)^{1/2}$ \quad $\|\mathbf{x}\|_1:=\sum_{i=1}^n|x_i|$

    \vspace{-5pt}
    \begin{minipage}[t]{0.45\textwidth}
        \begin{enumerate}
            \item $\|\mathbf{x}\|_p\ge0$ \quad $\|\mathbf{x}\|_p=0 \Leftrightarrow \mathbf{x}=\mathbf{0}$
            \item $\|\alpha\mathbf{x}\|_p=|\alpha|\|\mathbf{x}\|_p$ for $\alpha\in\R$
            \item $\|\mathbf{x}+\mathbf{y}\|_p\le\|\mathbf{x}\|_p+\|\mathbf{y}\|_p$ \quad \ps{(Triangle inequality)}
            \item \ok{Induced Norm}: $\|\mathbf{A}\|_p:=\sup\limits_{\mathbf{x}\ne0}\frac{\|\mathbf{A}\mathbf{x}\|_p}{\|\mathbf{x}\|_p}=\sup\limits_{\|\mathbf{x}\|_p=1}\|\mathbf{A}\mathbf{x}\|_p$
        \end{enumerate}    
    \end{minipage}
    \hfill
    \begin{minipage}[t]{0.55\textwidth}
        \begin{enumerate}
            \item $\|\mathbf{A}_{m\times n}\|_\infty:=\max\limits_{1\le j\le m}\sum_{i=1}^n|a_{ij}|$ \quad \comment{(maximum row sum)}
            \item $\|\mathbf{A}_{m\times n}\|_1:=\max\limits_{1\le i\le n}\sum_{j=1}^m|a_{ij}|$ \quad \comment{(maximum column sum)}
            \item $\|\mathbf{A}_{m\times n}\|_2:=\sqrt{\rho(\mathbf{A}^\top\mathbf{A})}$ \quad \comment{$\rho(\mathbf{A}^\top\mathbf{A})=\max\{|\lambda|:\lambda \text{ is eigenvalue of } \mathbf{A}^\top\mathbf{A}\}$}
        \end{enumerate}
        
        \vspace{10pt}
        \qquad{\scriptsize \ok{For Matrix}: $\|\mathbf{A}\mathbf{x}\|_p\le\|\mathbf{A}\|_p\|\mathbf{x}\|_p$ \quad $\|\mathbf{A}\mathbf{B}\|_p\le\|\mathbf{A}\|_p\|\mathbf{B}\|_p$ \quad $\|\mathbf{A}^k\|_p\le\|\mathbf{A}\|_p^k$ \quad $\|\mathbf{A}+\mathbf{B}\|_p\leq \|\mathbf{A}\|_p+\|\mathbf{B}\|_p$}
    \end{minipage}
\end{MyTh}
\vspace{-5pt}

\section{Methods for Systems of Linear Equations (SLE)}

\vspace{-0.05cm}
\qquad \ \ {\scriptsize \ok{Ideal:} We want to solve SLE: $\mathbf{A}\mathbf{x}=\mathbf{b}$, where $\mathbf{A}_{n\times n}$ is invertible, $\mathbf{x},\mathbf{b}\in\R^n$}
\vspace{-0.1cm}

\subsection{Backward Stable | Condition Number | Error}

\begin{MyTh}{Backward Stable}
  An algorithm to solve SLE is Backward Stable if: \ \ $\exists \alpha(n)>0$ s.t. $\forall \mathbf{A}, \mathbf{b}$, the computed solution $\widehat{\mathbf{x}}$ satisfies:
  
  \hspace{2.85cm} $(\mathbf{A} + \Delta \mathbf{A})\widehat{\mathbf{x}} = \mathbf{b} + \Delta \mathbf{b}$ \ \ and  \ \ $\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p} + \frac{\|\Delta \mathbf{b}\|_p}{\|\mathbf{b}\|_p} \leq \alpha \varepsilon_m$ \quad \comment{$\varepsilon_m:$ machine epsilon.} \quad \ps{ps: $\alpha$可以和问题规模$n$相关,但不能和$\mathbf{A}$或$\mathbf{b}$相关.}
\end{MyTh}

\vspace{-2pt}
\begin{MyTh}{Condition Number ($\kappa_p(\mathbf{A})$)}
    $\kappa_p(\mathbf{A}):=$
    {\tiny
    $\begin{cases}
      \|\mathbf{A}\|_p\|\mathbf{A}^{-1}\|_p\geq1 & \text{if } \mathbf{A} \text{ is invertible} \\
      \infty & \text{otherwise}
    \end{cases}
    \qquad\begin{cases}
      \text{If $\kappa_p(\mathbf{A})$ is large (e.g. $\kappa_p(\mathbf{A})\ge10^{12}\approx\varepsilon_m^{-1}$)}, & \text{$\mathbf{A}$ is \textcolor{green}{ill-conditioned.}} \\
      \text{If $\kappa_p(\mathbf{A})\leq10^4$}, & \text{$\mathbf{A}$ is \textcolor{green}{well-conditioned.}}
    \end{cases}$
    }
    \qquad \ps{ps: $\kappa_p(\mathbf{A})$ 衡量矩阵奇异性}

    \begin{enumerate}
      \item For $\mathbf{A}_{n\times n}$ with real eigenvalues $|\lambda_1| \geq |\lambda_2| \geq \cdots \geq |\lambda_n|$ and corresponding eigenvectors $\mathbf{x}_1, \mathbf{x}_2, \ldots, \mathbf{x}_n$: \quad {\tiny \textcolor{red}{ps:对所有实对称矩阵都满足下面条件}}
      
      \qquad If $\mathbf{x}_1, \mathbf{x}_2, \ldots, \mathbf{x}_n$ are \comment{orthonormal set}. \ \ Then $\kappa_2(\mathbf{A}) = \|\mathbf{A}\|_2\|\mathbf{A}^{-1}\|_2 = \frac{|\lambda_1|}{|\lambda_n|}\geq1$
    \end{enumerate}
\end{MyTh}

\vspace{2pt}
\begin{MyTh}{Relative Error}
  $\frac{\|\widehat{\mathbf{x}} - \mathbf{x}\|_p}{\|\mathbf{x}\|_p}$ \qquad \textcolor{blue}{\textbf{Actual Error:}} $\|\widehat{\mathbf{x}} - \mathbf{x}\|_p$ \qquad \quad \ps{ps: Relative error gives more information than actual error.}

  \begin{enumerate}
    \item Let $\mathbf{Ax}=\mathbf{b}$ and $(\mathbf{A} + \Delta \mathbf{A})\widehat{\mathbf{x}} = \mathbf{b}$. \ \ If $\kappa_p(\mathbf{A})\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}<1$ \ \ then \ \ $\frac{\|\mathbf{x}-\widehat{\mathbf{x}}\|_p}{\|\mathbf{x}\|_p}\leq\frac{\kappa_p(\mathbf{A})}{1-\kappa_p(\mathbf{A})\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}}\cdot\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}$. \quad \ps{ps: $\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}$ \ \ $\frac{\|\Delta \mathbf{b}\|_p}{\|\mathbf{b}\|_p}$ 小代表 good stability.}
    \item Let $\mathbf{Ax}=\mathbf{b}$ and $(\mathbf{A} + \Delta \mathbf{A})\widehat{\mathbf{x}} = \mathbf{b} + \Delta\mathbf{b}$. \ \ If $\kappa_p(\mathbf{A})\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}<1$ \ \ then \ \ $\frac{\|\mathbf{x}-\widehat{\mathbf{x}}\|_p}{\|\mathbf{x}\|_p}\leq\frac{\kappa_p(\mathbf{A})}{1-\kappa_p(\mathbf{A})\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}}\cdot(\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}+\frac{\|\Delta \mathbf{b}\|_p}{\|\mathbf{b}\|_p})$.
  \end{enumerate}
\end{MyTh}


\subsection{LU Factorization | Gaussian Elimination (GE)}

\begin{MyTh}{Def of LU}
    If $\mathbf{A}_{n\times n}$ is invertible, then $^1\exists \ \mathbf{L}$: \comment{单位下三角}, $^2\exists \ \mathbf{U}$: \comment{可逆上三角}. \ s.t. \ $\mathbf{A}=\mathbf{L}\mathbf{U}$ \ok{(unique)}
\end{MyTh}

\begin{MyTh}{LU Factorization}
    By Gaussian elimination, $\mathbf{A}_{n\times n}=(\prod_{k=1}^{n-1}\mathbf{L}_k)^{-1}\mathbf{U}=:\mathbf{L}\mathbf{U}$ \ \ \ps{$\mathbf{L}_k$: Lower triangular with 1s on the diagonal (by row operation on single column)}
    
    \begin{enumerate}
        \item 如果$\mathbf{L}$是\comment{单位下三角}矩阵,且只有第$k$列有非零元素在对角线下,则$\mathbf{L}^{-1}$矩阵是$\mathbf{L}$对角线不动,其余为相反数.
        \item 如果$\mathbf{L}_1,\mathbf{L}_2$矩阵是\comment{单位下三角}矩阵,且$\mathbf{L}_1|\mathbf{L}_2$仅在第$1\sim k|k+1\sim n$列有非零元素在对角线下,则$\mathbf{L}_1\mathbf{L}_2$矩阵是他们的"拼接".
        \item By LU factorization, $\mathbf{A}=\mathbf{L}\mathbf{U}$ \ \ $\Rightarrow$ \ \ By solving \ok{$\mathbf{L}\mathbf{y}=\mathbf{b}$} and \ok{$\mathbf{U}\mathbf{x}=\mathbf{y}$}, we can solve SLE \ok{$\mathbf{A}\mathbf{x}=\mathbf{b}$}. \quad \ps{用$\mathbf{FS}$和$\mathbf{BS}$算法}
    \end{enumerate}
    \vspace{5pt}
\end{MyTh}

\begin{MyTh}{Error Analysis of GE}
  Gaussian elimination is an \ok{unstable} algorithm. \ps{(i.e. floating point arithmetic can have a disastrous effect on the computed solution.)}
\end{MyTh}


\subsection{LUPP | LUCP | GEPP | GECP}

\begin{MyTh}{Permutation Matrix}
    For a permutation $\pi$, $\mathbf{P}_\pi:=[\mathbf{e}_{\pi(1)}, \dots, \mathbf{e}_{\pi(n)}]$ \quad \ps{\tiny ps: 写的时候是对$I$,第$i$行的1移动到第$\pi(i)$行. \ \ 注意: 用另一方式定义的$\mathbf{P}_\pi$会让后面对$\pi$结论不一样}
    
    \quad $\cdot$ \ok{左边乘$\mathbf{P}_\pi$: 交换行} \ps{($\mathbf{P}_\pi$的行 \ 代表具体行的交换方法, $\pi$也代表交换行的方法)} \qquad $\cdot$ \ok{右边乘$\mathbf{P}_\pi$: 交换列} \ps{($\mathbf{P}_\pi$的列 \ 代表具体列的交换方法, $\pi^{-1}$才代表交换列的方法)}

    \quad $\cdot$ Proposition: $\mathbf{P}_\pi$ is orthogonal, i.e. $\mathbf{P}_\pi^\top=\mathbf{P}_\pi^{-1}$
    \quad $\cdot$ \ok{对于二阶置换矩阵(Involution)}: $\mathbf{P}_\pi^2=\mathbf{I}$, i.e. $\mathbf{P}_\pi^{-1}=\mathbf{P}_\pi$
    \quad $\cdot$ \ok{GEPP|GECP}: {\tiny \ps{交换行(行和列)使每次$|u_{kk}|$最大\ok{(stable)}}}
\end{MyTh}

\begin{MyTh}{Backward Stable of GEPP}
  $\forall$ SLE sol $\widehat{\mathbf{x}}$ by GEPP, \ \ $\exists \Delta \mathbf{A}$ s.t. $(\mathbf{A} + \Delta \mathbf{A})\widehat{\mathbf{x}} = \mathbf{b}$ \ \ and \ \ $\frac{\|\Delta \mathbf{A}\|_{\infty}}{\|\mathbf{A}\|_{\infty}} \leq p(n) 2^{n-1}\varepsilon_m$ \ps{$p(n)$ is a cubic polynomial}
\end{MyTh}


\subsection{QR Factorization Method}

\begin{MyTh}{QR Factorization}
  QR factorization of $\mathbf{A}_{n\times n}$ is $\mathbf{A} = \mathbf{Q}\mathbf{R}$, where $\mathbf{Q}$ is \comment{orthogonal (i.e. $\mathbf{Q}^T=\mathbf{Q}^{-1}$)} and $\mathbf{R}$ is \comment{non-singular upper triangular.}

  \quad $\cdot$ Def of $\mathbf{Q}$ is orthogonal is equivalent to $\mathbf{Q}^T\mathbf{Q}=\mathbf{I}$. \comment{(i.e. columns of $\mathbf{Q}$ forming an orthonormal set.)}
  \quad $\cdot$ QR produce \ok{large errors}

  \quad $\cdot$ \begin{MyTh}{Orthogonal Projection}
    $\text{proj}_{\mathbf{u}}(\mathbf{x}) = \frac{\mathbf{x} \cdot \mathbf{u}}{\|\mathbf{u}\|^2_2} \mathbf{u}$ \ \comment{$=(\mathbf{u}^\top\mathbf{x})\mathbf{u}$ if $\|\mathbf{u}\|_2=1$.} \hspace{2cm} $\mathbf{x}=\text{proj}_{\mathbf{u}}(\mathbf{x})^{\text{parallel to }\mathbf{u}} + (\mathbf{x} - \text{proj}_{\mathbf{u}}(\mathbf{x}))^{\text{orthogonal to }\mathbf{u}}$
  \end{MyTh}

\end{MyTh}

\begin{MyTh}{Error Analysis of MGS}
  The matrices $\widehat{\mathbf{Q}}$ and $\widehat{\mathbf{Q}}$ computed by MGS satisfies: \quad $\widehat{\mathbf{Q}}^\top\widehat{\mathbf{Q}}=\mathbf{I}+\Delta\mathbf{I}$ \ \ $\widehat{\mathbf{Q}}\widehat{\mathbf{R}}=\mathbf{A}+\Delta\mathbf{A}$ where:

  \hspace{3.45cm} $\| \Delta\mathbf{I}\|_2 \leq \frac{\alpha_1(n) \varepsilon_m\kappa_2(\mathbf{A})}{1-\alpha_1(n)\varepsilon_m\kappa_2(\mathbf{A})}$, if $1-\alpha_1(n)\varepsilon_m\kappa_2(\mathbf{A})>0$ \qquad $\|\Delta\mathbf{A}\|_2\leq\alpha_2(n)\varepsilon_m\|\mathbf{A}\|_2$ \ps{ps: $\alpha_1(n),\alpha_2(n)$ 只由$n$决定}
\end{MyTh}


\subsection{Householder QR factorisation}

\begin{MyTh}{Householder QR}
  \comment{(Most)} \ok{stable} algorithm \qquad \textbf{Householder Reflection Matrix}: $\mathbf{H}(\mathbf{w}):=\mathbf{I}-2\frac{\mathbf{w}\mathbf{w}^\top}{\|\mathbf{w}\|_2^2}$ \quad \ok{$\mathbf{H}(\mathbf{w})\mathbf{x}=\mathbf{x}-2\text{proj}_{\mathbf{w}}(\mathbf{x})$}

  \vspace{-2pt}
  \quad $\cdot$ $\mathbf{H}(\mathbf{w})$ is \ok{orthogonal} and \ok{symmetric} \comment{(In the Algorithm is $\mathbf{Q}_k,\mathbf{H}_k$)} \qquad $\cdot$ $\mathbf{H}(\mathbf{w})\mathbf{x}$ \comment{\tiny 作用为让$\mathbf{x}$向量对\textit{$\mathbf{w}$的法向量(超平面)} 上进行反射}
\end{MyTh}

\begin{MyTh}{Error Analysis of Householder QR}
    Let $\widehat{\mathbf{x}}$ denote sol of $\mathbf{Ax}=\mathbf{b}$ by Householder QR+SQR. \quad Then $(\mathbf{A}+\Delta\mathbf{A})=\mathbf{b}$, where $\|\Delta\mathbf{A}\|_2\leq \alpha n^{\frac{5}{2}}\varepsilon_m\|\mathbf{A}\|_2$ \ps{$\alpha$ small constant} 
\end{MyTh}


\section{Iterative Methods for SLE}

\begin{MyTh}{Iterative Method}
  For SLE $\mathbf{Ax}=\mathbf{b}$, if $\mathbf{A}=\mathbf{M}+\mathbf{N}$ with $\mathbf{M}$ is \textit{simpler structure} than $\mathbf{A}$. \ps{\tiny (例如: 对角/三角/对称矩阵)} \ \ $\Rightarrow$ \ \ {\scriptsize 迭代求解近似解} \ \ \ps{\tiny Diff choices $\mathbf{M},\mathbf{N}\to$ different iterative methods.}
\end{MyTh}

\begin{MyTh}{Error Analysis}
  Error $\mathbf{e}_k:=\mathbf{x}-\mathbf{x}_k$ $\Rightarrow$ $\mathbf{e}_k=\mathbf{R}^k\mathbf{e}_0$ \ \ps{\tiny $\mathbf{R}:=-\mathbf{M}^{-1}\mathbf{N}$} \quad \ok{Theorem}: If $\|\mathbf{R}\|_p<1$, then $\lim\limits_{k\to\infty}\|\mathbf{e}_k\|_p=0$ \ \ $\Leftrightarrow$ \ \ $\lim\limits_{k\to\infty}\mathbf{x}_k=\mathbf{x}$ \ps{\tiny(Convergence)} \quad \ok{\scriptsize GAU 收敛快于 JAC}

  \vspace{-2pt}
  \quad \ok{JAC|GAU}: If $\mathbf{A}$ is \ok{strictly diagonally dominant} [i.e. $|a_{ii}|>\sum_{j\ne i}|a_{ij}|$ $\forall i$ \ps{\tiny (对角线元素大于该行其他元素之和)}], then converges for $p=\infty$. \quad \ps{\tiny For $p=1$ 看列和/转置.}
\end{MyTh}

\subsection{DP MV MM}

\begin{multicols}{3}

% ----- Algorithm DP -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{DP} Dot product \quad \ps{$C = \sum_{i=1}^{n} 2 = 2n=\Ocal(n)$}}
\textbf{Input:} $\mathbf{x}, \mathbf{y} \in \mathbb{R}^n$ \\
\textbf{Output:} $s = \mathbf{x}\cdot\mathbf{y} = \mathbf{x}^\top \mathbf{y}$
\begin{algorithmic}[1]
  \State $s = 0$
  \For{$i = 1, \dots, n$}
    \State $s = s + x_i y_i$
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm MV -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{MV} Matrix-vector multiplication \vspace{-0.1cm} \\ \ps{$C = \sum_{i=1}^{m} 2n = 2nm=\Ocal(mn)$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{m \times n}$, $\mathbf{x} \in \mathbb{R}^n$ \\
\textbf{Output:} $\mathbf{y} = \mathbf{A}\mathbf{x}$
\begin{algorithmic}[1]
  \For{$i = 1, \dots, m$}
    \State $y_i = \mathbf{a}_i^\top \mathbf{x}$ \codealg{DP}
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm MM -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{MM} Matrix-matrix multiplication \vspace{-0.1cm} \\ \ps{$C = \sum_{i=1}^{m} \sum_{j=1}^{p} 2n = 2mnp=\Ocal(mnp)$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{m \times n}$, $\mathbf{B} \in \mathbb{R}^{n \times p}$ \\
\textbf{Output:} $\mathbf{C} = \mathbf{A} \mathbf{B}$
\begin{algorithmic}[1]
  \For{$i = 1, \dots, m$}
    \For{$j = 1, \dots, p$}
      \State $c_{ij} = \mathbf{a}_i^\top \mathbf{b}_j$ \codealg{DP}
    \EndFor
  \EndFor
\end{algorithmic}
\end{algorithm}

\end{multicols}

\subsection{LU FS BS | GE GEPP GECP | LUPP LUCP}
\vspace{-9pt}

\begin{multicols}{3}

% ----- Algorithm LU -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{LU} LU factorisation \vspace{-0.1cm} \\ \ps{$C(n) = \sum^{n-1}_{k=1}(n-k)(1+2(n-k+1)) = \frac{2}{3}n^3+\frac{1}{2}n^2-\frac{7}{6}n$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{n \times n}$ \\
\textbf{Output:} $\mathbf{L}, \mathbf{U} \in \mathbb{R}^{n \times n}$ \ \ s.t. \ \ $\mathbf{A} = \mathbf{LU}$
\begin{algorithmic}[1]
  \State $\mathbf{L} = \mathbf{I},\ \mathbf{U} = \mathbf{A}$
  \For{$k = 1, \dots, n - 1$}
    \For{$j = k + 1, \dots, n$}
      \State $l_{jk} = u_{jk} / u_{kk}$
      \State $(u_{jk}, \dots, u_{jn}) = (u_{jk}, \dots, u_{jn}) - l_{jk}(u_{kk}, \dots, u_{kn})$
    \EndFor
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm FS -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{FS} Forward substitution \vspace{-0.1cm} \\ \ps{$C(n) = \sum_{j=1}^{n}[2(j-1)+1]=n^2=\Ocal(n^2)$}}
\textbf{Input:} $\mathbf{L} \in \mathbb{R}^{n \times n}$ (lower triangular), $\mathbf{b} \in \mathbb{R}^n$ \\
\textbf{Output:} $\mathbf{y} \in \mathbb{R}^n$ such that $\mathbf{Ly} = \mathbf{b}$
\begin{algorithmic}[1]
  \For{$j = 1, \dots, n$}
    \State $y_j = \frac{1}{l_{jj}} \left( b_j - \sum_{k = 1}^{j - 1} l_{jk} y_k \right)$
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm BS -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{BS} Back substitution \vspace{-0.1cm} \\ \ps{$C(n) = \sum_{j=1}^n[2(n-j)+1]=n^2=\Ocal(n^2)$}}
\textbf{Input:} $\mathbf{U} \in \mathbb{R}^{n \times n}$ (upper triangular), $\mathbf{y} \in \mathbb{R}^n$ \\
\textbf{Output:} $\mathbf{x} \in \mathbb{R}^n$ such that $\mathbf{Ux} = \mathbf{y}$
\begin{algorithmic}[1]
  \For{$j = n, \dots, 1$}
    \State $x_j = \frac{1}{u_{jj}} \left( y_j - \sum_{k = j + 1}^{n} u_{jk} x_k \right)$
  \EndFor
\end{algorithmic}
\end{algorithm}

\end{multicols}

\vspace{-0.8cm}
\begin{multicols}{3}

% ----- Algorithm GE -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{GE} Gaussian elimination \vspace{-0.1cm} \\ \ps{$C(n) = \frac{2}{3}n^3 + \frac{5}{2}n^2 - \frac{7}{6}n=\Ocal(n^3)$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{n \times n},\ \mathbf{b} \in \mathbb{R}^n$ \\
\textbf{Output:} $\mathbf{x} \in \mathbb{R}^n$ \ \ s.t. \ \ $\mathbf{Ax} = \mathbf{b}$
\begin{algorithmic}[1]
  \State Find LU factorisation of $\mathbf{A}$. \codealg{LU}
  \State Solve $\mathbf{Ly} = \mathbf{b}$ using forward substitution. \codealg{FS}
  \State Solve $\mathbf{Ux} = \mathbf{y}$ using back substitution. \codealg{BS}
\end{algorithmic}
\end{algorithm}

% ----- Algorithm GEPP -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{GEPP} Gaussian elimination with partial pivoting \vspace{-0.1cm} \\ \ps{$C(n)=\frac23n^3+3n^2-\frac53n=\Ocal(n^3)$}}
\textbf{Input:} $\mathbf{A}\in\R^{n\times n},\ \mathbf b\in\R^n$ \textcolor{gray}{\quad \# numpy.linalg.solve use it} \\
\textbf{Output:} $\mathbf x\in\R^n$ satisfying $\mathbf{A x}=\mathbf b$
\begin{algorithmic}[1]
  \State Find $\mathbf P\mathbf A=\mathbf L\mathbf U$ using Algorithm LUPP \codealg{LUPP}
  \State Solve $\mathbf L\,\mathbf y = \mathbf P\,\mathbf b$ by forward substitution \codealg{FS}
  \State Solve $\mathbf U\,\mathbf x = \mathbf y$ by back substitution \codealg{BS}
\end{algorithmic}
\end{algorithm}

% ----- Algorithm GECP -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{GECP} Gaussian elimination with complete pivoting \vspace{-0.1cm} \\ \ps{$C(n)=n^3+3n^2-2n=\Ocal(n^3)$}}
\textbf{Input:} $\mathbf{A}\in\R^{n\times n},\ \mathbf b\in\R^n$ \\
\textbf{Output:} $\mathbf x\in\R^n$ satisfying $\mathbf A\,\mathbf x=\mathbf b$
\begin{algorithmic}[1]
  \State Find $\mathbf P\,\mathbf A\,\mathbf Q = \mathbf L\,\mathbf U$ using Algorithm LUCP \codealg{LUCP}
  \State Solve $\mathbf L\,\mathbf y = \mathbf P\,\mathbf b$ by forward substitution \codealg{FS}
  \State Solve $\mathbf U\,\mathbf z = \mathbf y$ by back substitution \codealg{BS}
  \State Compute $\mathbf x = \mathbf Q\,\mathbf z$
\end{algorithmic}
\end{algorithm}

\end{multicols}

\vspace{-0.8cm}
\begin{multicols}{2}

% ----- Algorithm LUPP -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{LUPP} LU factorisation with partial pivoting \ps{(not unique)} \vspace{-0.1cm} \\ \ps{$C(n)=\frac{2}{3}n^3+n^2-\frac{5}{3}n$}}
\textbf{Input:} $\mathbf{A}\in\R^{n\times n}$ \\
\textbf{Output:} $\mathbf{L},\ \mathbf{U},\ \mathbf{P}\in\R^{n\times n}$ \ s.t. \ $\mathbf{P}\mathbf{A}=\mathbf{L}\mathbf{U}$ with $\mathbf{L}$ 单位下三角, $\mathbf{U}$ 可逆上三角
\begin{algorithmic}[1]
  \State $\mathbf{U} = \mathbf{A},\ \mathbf{L} = \mathbf{I}$,\ $\mathbf{P} = \mathbf{I}$
  \For{$k = 1, \dots, n - 1$}
    \State Choose $i\in\{k,\dots,n\}$ which maximizes $|u_{ik}|$ \codealg{Max}
    \State Exchange row $(u_{kk}, \dots, u_{kn})$ with $(u_{ik}, \dots, u_{in})$ \textcolor{gray}{\quad \# 第$k$行和第$i$行 \ 的$k\sim n$列交换}
    \State Exchange row $(l_{k1}, \dots, l_{k,k-1})$ with $(l_{i1}, \dots, l_{i,k-1})$ \textcolor{gray}{\quad \# 第$k$行和第$i$行 \ 的$1\sim k-1$列交换}
    \State Exchange row $(p_{k1}, \dots, p_{kn})$ with $(p_{i1}, \dots, p_{in})$ \textcolor{gray}{\quad \# 第$k$行和第$i$行 \ 的全部列交换}
    \For{$j = k + 1, \dots, n$}
      \State $l_{jk} = u_{jk} / u_{kk}$
      \State $(u_{jk}, \dots, u_{jn}) = (u_{jk}, \dots, u_{jn}) - l_{jk}(u_{kk}, \dots, u_{kn})$
    \EndFor
  \EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\tiny
\caption*{\textbf{LUCP} LU factorisation with complete pivoting \ps{(not unique)} \vspace{-0.1cm} \\ \ps{$C(n)=n^3+n^2-2n$}}
\textbf{Input:} $\mathbf{A}\in\R^{n\times n}$ \\
\textbf{Output:} $\mathbf{L},\ \mathbf{U},\ \mathbf{P},\ \mathbf{Q}\in\R^{n\times n}$ \ s.t. \ $\mathbf{P}\mathbf{A}\mathbf{Q}=\mathbf{L}\mathbf{U}$ with $\mathbf{L}$ 单位下三角, $\mathbf{U}$ 可逆上三角
\begin{algorithmic}[1]
  \State $\mathbf{U} = \mathbf{A},\ \mathbf{L} = \mathbf{I},\ \mathbf{P} = \mathbf{I},\ \mathbf{Q} = \mathbf{I}$
  \For{$k = 1, \dots, n - 1$}
    \State Choose $(i,j)\in\{(k,k),...,(n,n)\}$ which maximizes $|u_{ij}|$ \codealg{Max}
    \State Exchange row $(u_{k k}, \dots, u_{k n})$ with $(u_{i k}, \dots, u_{i n})$ \textcolor{gray}{\quad \# 第$k$行和第$i$行 \ 的$k\sim n$列交换 $U$}
    \State Exchange row $(l_{k1}, \dots, l_{k,k-1})$ with $(l_{i1}, \dots, l_{i,k-1})$ \textcolor{gray}{\quad \# 第$k$行和第$i$行 \ 的$1\sim k-1$列交换 $L$}
    \State Exchange row $(p_{k1}, \dots, p_{kn})$ with $(p_{i1}, \dots, p_{in})$ \textcolor{gray}{\quad \# 第$k$行和第$i$行 \ 的全部列交换 $P$}
    \State Exchange column $(u_{1 k}, \dots, u_{n k})$ with $(u_{1 j}, \dots, u_{n j})$ \textcolor{gray}{\quad \# 第$k$列和第$j$列 \ 的全部行交换 $U$}
    \State Exchange column $(q_{1 k}, \dots, q_{n k})$ with $(q_{1 j}, \dots, q_{n j})$ \textcolor{gray}{\quad \# 第$k$列和第$j$列 \ 的全部行交换 $Q$}
    \For{$m = k + 1, \dots, n$}
      \State $l_{m k} = u_{m k} / u_{k k}$
      \State $(u_{m k}, \dots, u_{m n}) = (u_{m k}, \dots, u_{m n}) - l_{m k}(u_{k k}, \dots, u_{k n})$
    \EndFor
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm (M)GS -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{(M)GS} (Modified) Gram-Schmidt \quad \ps{$C(n)=2n^3+n^2+n$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{n \times n}$, with columns $\mathbf{a}_1, \dots, \mathbf{a}_n$ \\
\textbf{Output:} $\mathbf{Q} \in \mathbb{R}^{n \times n}$, with orthonormal columns $\mathbf{q}_1, \dots, \mathbf{q}_n$

\qquad\qquad $\mathbf{R} \in \mathbb{R}^{n \times n}$ non-singular upper triangular matrix; \ s.t. $\mathbf{A} = \mathbf{Q}\mathbf{R}$
\begin{algorithmic}[1]
  \State $\mathbf{R} = \mathbf{0}$
  \For{$j=1,...,n$}
    \State $\mathbf{q}_j = \mathbf{a}_j$
    \For{$k=1,...,j-1$}
      \State $r_{kj} = \mathbf{q}_k^\top \mathbf{a}_j$ \quad | \quad ($r_{kj} = \mathbf{q}_k^\top \mathbf{q}_j$)
      \State $\mathbf{q}_j = \mathbf{q}_j - r_{kj} \mathbf{q}_k$ 
    \EndFor
    \State $r_{jj} = \|\mathbf{q}_j\|_2=\sqrt{\mathbf{q}_j^\top \mathbf{q}_j}$
    \State $\mathbf{q}_j = \frac{\mathbf{q}_j}{r_{jj}}$
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm QR -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{QR} Householder QR factorization \quad \ps{$C(n)=\frac{8}{3}n^3+\Ocal(n^2)$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{n \times n}$ \\
\textbf{Output:} $\mathbf{Q} \in \mathbb{R}^{n \times n}$ orthogonal

\qquad\qquad $\mathbf{R} \in \mathbb{R}^{n \times n}$ non-singular upper triangular matrix; \ s.t. $\mathbf{A} = \mathbf{Q}\mathbf{R}$
\begin{algorithmic}[1]
  \State $\mathbf{Q} = \mathbf{I}$,\ $\mathbf{R} = \mathbf{A}$
  \For{$k = 1, \dots, n - 1$}
    \State $\mathbf{u} = (r_{kk}, \dots, r_{nk})\in \mathbb{R}^{n-k+1}$
    \State $\mathbf{v} = \mathbf{u} + \mathrm{sign}(u_1)\|\mathbf{u}\|_2 \mathbf{e}_1$  \quad \textcolor{gray}{\tiny \quad \ \# line.4-7 得到的$\mathbf{Q}_k$:}
    \State $\mathbf{w} = \frac{\mathbf{v}}{\|\mathbf{v}\|_2}$  \qquad\qquad\qquad\quad \textcolor{gray}{\tiny \qquad \# 使得$\mathbf{Q}_k\mathbf{u}=-\text{sign}(u_1)\|\mathbf{u}\|_2\mathbf{e}_1$}
    \State $\mathbf{H}_k = \mathbf{I} - 2\mathbf{w}\mathbf{w}^\top\in \mathbb{R}^{(n-k+1) \times (n-k+1)}$ \quad \textcolor{gray}{\scalebox{0.9}{\tiny \# line.6-8 一起算,见下}}
    \State $\mathbf{Q}_k = \mathbf{I}_{n-k} \oplus \mathbf{H}_k\in\R^{n\times n}$ \ \textcolor{gray}{\scalebox{0.75}{\tiny \# $\downarrow$ line.6-8 $C=4(n-k+1)^2+(n-k+1)$}}
    \State $\mathbf{R} = \mathbf{Q}_k \mathbf{R}$ \ \ \textcolor{gray}{\scalebox{0.8}{\tiny \# 用矩阵分块,只有右下角$\mathbf{H}_k\mathbf{R}_{k:n,k:n}$要算$\to$用$H_k$定义Alg. MV+OP}}
    \State $\mathbf{Q} = \mathbf{Q} \mathbf{Q}_k$ \ \ \textcolor{gray}{\scalebox{0.8}{\tiny \# If using in Alg. \imp{SQR}, then $9':$ $\mathbf{y}=\mathbf{Q}_k\mathbf{y}$; \quad $C=\frac{4}{3}n^3+\Ocal(n^2)$ (SQR/QR)}}
  \EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\tiny
\caption*{\textbf{\ok{GI}|JAC|GAU} \quad \ok{\tiny JAC:$C=\Ocal(k_{\max} n^2)$} \quad | \quad \ok{\tiny GAU:$C=\Ocal(k_{\max} n^2)$}}
\textbf{Input:} $\mathbf{A} = \mathbf{M} + \mathbf{N} \in \mathbb{R}^{n \times n}$, $\mathbf{b} \in \mathbb{R}^n$, $\mathbf{x}_0\in \mathbb{R}^n$, $\eps_r>0$ \\
\textbf{Output:} $\mathbf{x}_k \in \mathbb{R}^n$ with $\mathbf{Ax}_k \approx \mathbf{b}$
\begin{algorithmic}[1]
  \State $k = 0$ \ \ \textcolor{gray}{\tiny \# JAC: $C=2n^2$/it \quad GAU: $C=2n^2$/it}
  \While{$\|\mathbf{Ax}_k - \mathbf{b}\|_p > \eps_r $} \quad \textcolor{gray}{\tiny \# e.g. $\eps_r=10^{-3}\|\mathbf{b}\|_p$}
    \State $k = k + 1$ \quad \textcolor{green}{\tiny \# 上面给的$C$不含这步}
    \State Compute $\mathbf{y}_k = \mathbf{b}-\mathbf{N}\mathbf{x}_{k-1}$
    \State Solve $\mathbf{M}\mathbf{x}_k = \mathbf{y}_k$ \quad \textcolor{gray}{\tiny \# $\downarrow$ $\mathbf{A}=\mathbf{D}+\mathbf{L}+\mathbf{U}$}
  \EndWhile \quad \textcolor{red}{\tiny \# JAC: $M=D$, $N=L+U$; \ GAU:$M=D+L$, $N=U$}
\end{algorithmic}
\end{algorithm}

    \EndFor
  \EndFor
\end{algorithmic}
\end{algorithm}

\end{MyTh}

        \item \textbf{MGS vs. GS}:In floating point arithmetic, the computed vectors $\{\hat{q}_1, . . . , \hat{q}_{j−1}\}$ will not be orthonormal. Algorithm MGS takes this into account when computing $\hat{q}_j$; Algorithm GS does not.
