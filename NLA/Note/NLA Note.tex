\section{Background}

\begin{MyTh}{Floating Point ($\F$)}
    A number = mantissa $\times$ base$^{\text{exponent}}$ \quad \comment{\scriptsize (e.g. $2048=1\times 2^{11}$ \ps{mantissa=1, base=2, exponent=11})}
\end{MyTh}

\begin{MyTh}{Assumptions}
    \comment{There is $\varepsilon_m>0$ \ok{(machine epsilon)}} \ \ \ok{A1}: $\forall\alpha\in\R$, $\exists \varepsilon\in(-\varepsilon_m,\varepsilon_m)$ s.t. $fl:\R\to\F$ by $fl(\alpha)=\alpha(1+\varepsilon)$
    
    \qquad\qquad\qquad\quad {\small $\varepsilon_m\approx 2.22\times 10^{-16}$} \hspace{65pt} \ok{A2}: $\forall\alpha,\beta\in\F$, $\forall *\in\{+,-,\times,\div\}$, $\exists \varepsilon\in(-\varepsilon_m,\varepsilon_m)$ s.t. $\alpha \circledast \beta= (\alpha* \beta)(1+\varepsilon)$

    \vspace{-0.09cm}\hspace{2cm}\ps{A1: The rounding error in $\alpha$ is relative to the size of $\alpha$ ($|\alpha-fl(\alpha)|<|\alpha|\varepsilon_m$) \quad A2: The error in $\alpha \circledast \beta$ is relative to the size of $\alpha * \beta$ ($|\alpha \circledast \beta - \alpha * \beta|\le|\alpha * \beta|\varepsilon_m$)}
\end{MyTh}

\begin{MyTh}{Cost of Algorithm ($C$)}
    $C:=$ number of floating point operations (FLOPs) \qquad If $C(n)\le\alpha f(n)$, $\Rightarrow$ $C(n)=\Ocal(f(n))$.

    \quad We consider that: \ok{交换/赋值/相反数} no cost \ps{(i.e. Permutation $\mathbf{P}$ is free)}, \ \ \ok{比较两数} 1 FLOPs, \ \ \ok{$\sqrt{~~~~}$} 1 FLOPs.
\end{MyTh}

\begin{MyTh}{Norms}
    $\|\mathbf{x}\|_p:=(\sum_{i=1}^n|x_i|^p)^{1/p}$, $p\ge1$ \quad $\|\mathbf{x}\|_\infty:=\max_{1\le i\le n}|x_i|$ \quad $\|\mathbf{x}\|_2:=(\sum_{i=1}^n|x_i|^2)^{1/2}$ \quad $\|\mathbf{x}\|_1:=\sum_{i=1}^n|x_i|$

    \vspace{-5pt}
    \begin{minipage}[t]{0.45\textwidth}
        \begin{enumerate}
            \item $\|\mathbf{x}\|_p\ge0$ \quad $\|\mathbf{x}\|_p=0 \Leftrightarrow \mathbf{x}=\mathbf{0}$
            \item $\|\alpha\mathbf{x}\|_p=|\alpha|\|\mathbf{x}\|_p$ for $\alpha\in\R$
            \item $\|\mathbf{x}+\mathbf{y}\|_p\le\|\mathbf{x}\|_p+\|\mathbf{y}\|_p$ \quad \ps{(Triangle inequality)}
            \item \ok{Induced Norm}: $\|\mathbf{A}\|_p:=\sup\limits_{\mathbf{x}\ne0}\frac{\|\mathbf{A}\mathbf{x}\|_p}{\|\mathbf{x}\|_p}=\sup\limits_{\|\mathbf{x}\|_p=1}\|\mathbf{A}\mathbf{x}\|_p$
        \end{enumerate}    
    \end{minipage}
    \hfill
    \begin{minipage}[t]{0.55\textwidth}
        \begin{enumerate}
            \item $\|\mathbf{A}_{m\times n}\|_\infty:=\max\limits_{1\le j\le m}\sum_{i=1}^n|a_{ij}|$ \quad \comment{(maximum row sum)}
            \item $\|\mathbf{A}_{m\times n}\|_1:=\max\limits_{1\le i\le n}\sum_{j=1}^m|a_{ij}|$ \quad \comment{(maximum column sum)}
            \item $\|\mathbf{A}_{m\times n}\|_2:=\sqrt{\rho(\mathbf{A}^\top\mathbf{A})}$ \quad \comment{$\rho(\mathbf{A}^\top\mathbf{A})=\max\{|\lambda|:\lambda \text{ is eigenvalue of } \mathbf{A}^\top\mathbf{A}\}$}
        \end{enumerate}
        
        \vspace{10pt}
        \qquad{\scriptsize \ok{For Matrix}: $\|\mathbf{A}\mathbf{x}\|_p\le\|\mathbf{A}\|_p\|\mathbf{x}\|_p$ \quad $\|\mathbf{A}\mathbf{B}\|_p\le\|\mathbf{A}\|_p\|\mathbf{B}\|_p$ \quad $\|\mathbf{A}^k\|_p\le\|\mathbf{A}\|_p^k$ \quad $\|\mathbf{A}+\mathbf{B}\|_p\leq \|\mathbf{A}\|_p+\|\mathbf{B}\|_p$}
    \end{minipage}
\end{MyTh}
\vspace{-5pt}

\section{Methods for Systems of Linear Equations (SLE)}

\vspace{-0.05cm}
\qquad \ \ {\scriptsize \ok{Ideal:} We want to solve SLE: $\mathbf{A}\mathbf{x}=\mathbf{b}$, where $\mathbf{A}_{n\times n}$ is invertible, $\mathbf{x},\mathbf{b}\in\R^n$}
\vspace{-0.1cm}

\subsection{Backward Stable | Condition Number | Error}

\begin{MyTh}{Backward Stable}
  \scriptsize
  Algorithm solve SLE is Backward Stable if: \ \ $\exists \alpha(n)>0$ s.t. $\forall \mathbf{A}, \mathbf{b}$, computed solution $\widehat{\mathbf{x}}$ has:
  \quad $(\mathbf{A} + \Delta \mathbf{A})\widehat{\mathbf{x}} = \mathbf{b} + \Delta \mathbf{b}$ \ \ and  \ \ $\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p} + \frac{\|\Delta \mathbf{b}\|_p}{\|\mathbf{b}\|_p} \leq \alpha \varepsilon_m$ \quad \ps{$\varepsilon_m:$ machine epsilon.}
\end{MyTh}

\vspace{-4pt}
\begin{MyTh}{Condition Number ($\kappa_p(\mathbf{A})$)}
    $\kappa_p(\mathbf{A}):=$
    {\tiny
    $\begin{cases}
      \|\mathbf{A}\|_p\|\mathbf{A}^{-1}\|_p\geq 1& \text{if } \mathbf{A} \text{ is invertible} \\
      \infty & \text{otherwise}
    \end{cases}
    \qquad\begin{cases}
      \text{If $\kappa_p(\mathbf{A})$ is large (e.g. $\kappa_p(\mathbf{A})\ge10^{12}\approx\varepsilon_m^{-1}$)}, & \text{$\mathbf{A}$ is \textcolor{green}{ill-conditioned.}} \\
      \text{If $\kappa_p(\mathbf{A})\leq10^4$}, & \text{$\mathbf{A}$ is \textcolor{green}{well-conditioned.}}
    \end{cases}$
    }
    \qquad \ps{ps: $\kappa_p(\mathbf{A})$ 衡量矩阵奇异性}

    \vspace{-5pt}
    {\quad \footnotesize If $\mathbf{A}_{n\times n}$ s.t. $|\lambda_1| \geq |\lambda_2| \geq \cdots \geq |\lambda_n|$ corresponding $\mathbf{x}_1, \mathbf{x}_2, \ldots, \mathbf{x}_n$ are \ps{orthonormal set}. \imp{\scriptsize (e.g. $\overline{A}^\top=A$)} \quad Then $\kappa_2(\mathbf{A}) = \|\mathbf{A}\|_2\|\mathbf{A}^{-1}\|_2 = \frac{|\lambda_1|}{|\lambda_n|}\geq1$}
\end{MyTh}

\vspace{-3pt}
\begin{MyTh}{Relative Error}
  {\footnotesize $\frac{\|\widehat{\mathbf{x}} - \mathbf{x}\|_p}{\|\mathbf{x}\|_p}$ \ \ \textcolor{blue}{\textbf{Actual Error:}} $\|\widehat{\mathbf{x}} - \mathbf{x}\|_p$ \quad Let $\mathbf{Ax}=\mathbf{b}$ and $(\mathbf{A} + \Delta \mathbf{A})\widehat{\mathbf{x}} = \mathbf{b}$. \ \ If $\kappa_p(\mathbf{A})\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}<1$, \ \ then: \ \ {\tiny $\frac{\|\mathbf{x}-\widehat{\mathbf{x}}\|_p}{\|\mathbf{x}\|_p}\leq\frac{\kappa_p(\mathbf{A})}{1-\kappa_p(\mathbf{A})\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}}\cdot\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}$}}
  
  \vspace{-8pt}
  {\quad \scriptsize Generally, Let $\mathbf{Ax}=\mathbf{b}$ and $(\mathbf{A} + \Delta \mathbf{A})\widehat{\mathbf{x}} = \mathbf{b} + \Delta\mathbf{b}$. \ \ If $\kappa_p(\mathbf{A})\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}<1$ \ \ then \ \ $\frac{\|\mathbf{x}-\widehat{\mathbf{x}}\|_p}{\|\mathbf{x}\|_p}\leq\frac{\kappa_p(\mathbf{A})}{1-\kappa_p(\mathbf{A})\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}}\cdot(\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}+\frac{\|\Delta \mathbf{b}\|_p}{\|\mathbf{b}\|_p})$} \qquad\qquad\qquad\qquad\qquad\quad \ps{\tiny ps: $\frac{\|\Delta \mathbf{A}\|_p}{\|\mathbf{A}\|_p}$ \ \ $\frac{\|\Delta \mathbf{b}\|_p}{\|\mathbf{b}\|_p}$ 小代表 good stability.}
\end{MyTh}
\vspace{-7pt}


\subsection{LU Factorization | Gaussian Elimination (GE)}

\begin{MyTh}{Def of LU}
    If $\mathbf{A}_{n\times n}$ is invertible, then $^1\exists \ \mathbf{L}$: \comment{单位下三角}, $^2\exists \ \mathbf{U}$: \comment{可逆上三角}. \ s.t. \ $\mathbf{A}=\mathbf{L}\mathbf{U}$ \ok{(unique)}
\end{MyTh}

\begin{MyTh}{LU Factorization}
    By Gaussian elimination, $\mathbf{A}_{n\times n}=(\prod_{k=1}^{n-1}\mathbf{L}_k)^{-1}\mathbf{U}=:\mathbf{L}\mathbf{U}$ \ \ \ps{$\mathbf{L}_k$: Lower triangular with 1s on the diagonal (by row operation on single column)}
    
    \begin{enumerate}
        \item 如果$\mathbf{L}$是\comment{单位下三角}矩阵,且只有第$k$列有非零元素在对角线下,则$\mathbf{L}^{-1}$矩阵是$\mathbf{L}$对角线不动,其余为相反数.
        \item 如果$\mathbf{L}_1,\mathbf{L}_2$矩阵是\comment{单位下三角}矩阵,且$\mathbf{L}_1|\mathbf{L}_2$仅在第$1\sim k|k+1\sim n$列有非零元素在对角线下,则$\mathbf{L}_1\mathbf{L}_2$矩阵是他们的"拼接".
    \end{enumerate}
    \vspace{5pt}
\end{MyTh}

\begin{MyTh}{Error Analysis of GE}
  Gaussian elimination is an \ok{unstable} algorithm. \ps{(i.e. floating point arithmetic can have a disastrous effect on the computed solution.)}
\end{MyTh}


\subsection{LUPP | LUCP | GEPP | GECP}

\begin{MyTh}{Permutation Matrix}
    For a permutation $\pi$, $\mathbf{P}_\pi:=[\mathbf{e}_{\pi(1)}, \dots, \mathbf{e}_{\pi(n)}]$ \quad \ps{\tiny ps: 写的时候是对$I$,第$i$行的1移动到第$\pi(i)$行. \ \ 注意: 用另一方式定义的$\mathbf{P}_\pi$会让后面对$\pi$结论不一样}
    
    \quad $\cdot$ \ok{左边乘$\mathbf{P}_\pi$: 交换行} \ps{($\mathbf{P}_\pi$的行 \ 代表具体行的交换方法, $\pi$也代表交换行的方法)} \qquad $\cdot$ \ok{右边乘$\mathbf{P}_\pi$: 交换列} \ps{($\mathbf{P}_\pi$的列 \ 代表具体列的交换方法, $\pi^{-1}$才代表交换列的方法)}

    \quad $\cdot$ Proposition: $\mathbf{P}_\pi$ is orthogonal, i.e. $\mathbf{P}_\pi^\top=\mathbf{P}_\pi^{-1}$
    \quad $\cdot$ \ok{对于二阶置换矩阵(Involution)}: $\mathbf{P}_\pi^2=\mathbf{I}$, i.e. $\mathbf{P}_\pi^{-1}=\mathbf{P}_\pi$
    \quad $\cdot$ \ok{GEPP|GECP}: {\tiny \ps{交换行(行和列)使每次$|u_{kk}|$最大\ok{(stable)}}}
\end{MyTh}

\begin{MyTh}{Backward Stable of GEPP}
  $\forall$ SLE sol $\widehat{\mathbf{x}}$ by GEPP, \ \ $\exists \Delta \mathbf{A}$ s.t. $(\mathbf{A} + \Delta \mathbf{A})\widehat{\mathbf{x}} = \mathbf{b}$ \ \ and \ \ $\frac{\|\Delta \mathbf{A}\|_{\infty}}{\|\mathbf{A}\|_{\infty}} \leq p(n) 2^{n-1}\varepsilon_m$ \ps{$p(n)$ is a cubic polynomial}
\end{MyTh}


\subsection{QR Factorization Method}

\begin{MyTh}{QR Factorization}
  QR factorization of $\mathbf{A}_{n\times n}$ is $\mathbf{A} = \mathbf{Q}\mathbf{R}$, where $\mathbf{Q}$ is \comment{orthogonal (i.e. $\mathbf{Q}^T=\mathbf{Q}^{-1}$)} and $\mathbf{R}$ is \comment{non-singular upper triangular.}

  \quad $\cdot$ Def of $\mathbf{Q}$ is orthogonal is equivalent to $\mathbf{Q}^T\mathbf{Q}=\mathbf{I}$. \comment{(i.e. columns of $\mathbf{Q}$ forming an orthonormal set.)}
  \quad $\cdot$ QR produce \ok{large errors}

  \quad $\cdot$ \begin{MyTh}{Orthogonal Projection}
    $\text{proj}_{\mathbf{u}}(\mathbf{x}) = \frac{\mathbf{x} \cdot \mathbf{u}}{\|\mathbf{u}\|^2_2} \mathbf{u}$ \ \comment{$=(\mathbf{u}^\top\mathbf{x})\mathbf{u}$ if $\|\mathbf{u}\|_2=1$.} \hspace{2cm} $\mathbf{x}=\text{proj}_{\mathbf{u}}(\mathbf{x})^{\text{parallel to }\mathbf{u}} + (\mathbf{x} - \text{proj}_{\mathbf{u}}(\mathbf{x}))^{\text{orthogonal to }\mathbf{u}}$
  \end{MyTh}

\end{MyTh}

\begin{MyTh}{Error Analysis of MGS}
  The matrices $\widehat{\mathbf{Q}}$ and $\widehat{\mathbf{Q}}$ computed by MGS satisfies: \quad $\widehat{\mathbf{Q}}^\top\widehat{\mathbf{Q}}=\mathbf{I}+\Delta\mathbf{I}$ \ \ $\widehat{\mathbf{Q}}\widehat{\mathbf{R}}=\mathbf{A}+\Delta\mathbf{A}$ where:

  \hspace{3.0cm} $\| \Delta\mathbf{I}\|_2 \leq \frac{\alpha_1(n) \varepsilon_m\kappa_2(\mathbf{A})}{1-\alpha_1(n)\varepsilon_m\kappa_2(\mathbf{A})}$, if $1-\alpha_1(n)\varepsilon_m\kappa_2(\mathbf{A})>0$ \qquad $\|\Delta\mathbf{A}\|_2\leq\alpha_2(n)\varepsilon_m\|\mathbf{A}\|_2$ \ps{ps: $\alpha_1(n),\alpha_2(n)$ 只由$n$决定}
\end{MyTh}


\subsection{Householder QR factorisation}

\begin{MyTh}{Householder QR}
  \comment{(Most)} \ok{stable} algorithm \qquad \textbf{Householder Reflection Matrix}: $\mathbf{H}(\mathbf{w}):=\mathbf{I}-2\frac{\mathbf{w}\mathbf{w}^\top}{\|\mathbf{w}\|_2^2}$ \quad \ok{$\mathbf{H}(\mathbf{w})\mathbf{x}=\mathbf{x}-2\text{proj}_{\mathbf{w}}(\mathbf{x})$}

  \vspace{-2pt}
  \quad $\cdot$ $\mathbf{H}(\mathbf{w})$ is \ok{orthogonal} and \ok{symmetric} \comment{(In the Algorithm is $\mathbf{Q}_k,\mathbf{H}_k$)} \qquad $\cdot$ $\mathbf{H}(\mathbf{w})\mathbf{x}$ \comment{\tiny 作用为让$\mathbf{x}$向量对\textit{$\mathbf{w}$的法向量(超平面)} 上进行反射}
\end{MyTh}

\begin{MyTh}{Error Analysis of Householder QR}
    Let $\widehat{\mathbf{x}}$ denote sol of $\mathbf{Ax}=\mathbf{b}$ by Householder QR+SQR. \quad Then $(\mathbf{A}+\Delta\mathbf{A})=\mathbf{b}$, where $\|\Delta\mathbf{A}\|_2\leq \alpha n^{\frac{5}{2}}\varepsilon_m\|\mathbf{A}\|_2$ \ps{$\alpha$ small constant} 
\end{MyTh}


\subsection{Iterative Methods for SLE}

\begin{MyTh}{Iterative Method}
  For SLE $\mathbf{Ax}=\mathbf{b}$, if $\mathbf{A}=\mathbf{M}+\mathbf{N}$ with $\mathbf{M}$ is \textit{simpler structure} than $\mathbf{A}$. \ps{\tiny (例如: 对角/三角/对称矩阵)} \ \ $\Rightarrow$ \ \ {\scriptsize 迭代求解近似解} \ \ \ps{\tiny Diff choices $\mathbf{M},\mathbf{N}\to$ different iterative methods.}
\end{MyTh}

\begin{MyTh}{Error Analysis}
  Error $\mathbf{e}_k:=\mathbf{x}-\mathbf{x}_k$ $\Rightarrow$ $\mathbf{e}_k=\mathbf{R}^k\mathbf{e}_0$ \ \ps{\tiny $\mathbf{R}:=-\mathbf{M}^{-1}\mathbf{N}$} \quad \ok{Theorem}: If $\|\mathbf{R}\|_p<1$, then $\lim\limits_{k\to\infty}\|\mathbf{e}_k\|_p=0$ \ \ $\Leftrightarrow$ \ \ $\lim\limits_{k\to\infty}\mathbf{x}_k=\mathbf{x}$ \ps{\tiny(Convergence)} \quad \ok{\scriptsize GAU 收敛快于 JAC}

  \vspace{-2pt}
  \quad \ok{JAC|GAU}: If $\mathbf{A}$ is \ok{strictly diagonally dominant} [i.e. $|a_{ii}|>\sum_{j\ne i}|a_{ij}|$ $\forall i$ \ps{\tiny (对角线元素大于该行其他元素之和)}], then converges for $p=\infty$. \quad \ps{\tiny For $p=1$ 看列和/转置.}
\end{MyTh}


\section{Polynomial Fitting \& LSQ \qquad {\tiny ps:范德蒙德矩阵ill-cond}}

\begin{MyTh}{Vandermonde Matrix}
  $A\in \R^{n\times m}$ s.t. $a_{ij}=a_i^{j-1}$
\end{MyTh}
\begin{MyTh}{Poly}
  Deg $m-1$ poly fit to points: $\{(a_i,b_i)\}_{i=1}^n$ by solving \textit{$\mathbf{Ax}=\mathbf{b}$} $\Rightarrow$ \ps{\tiny sol $\mathbf{x}$ is coeff. of poly $p(x)=\sum_{j=1}^m x_j x^{j-1}$}
\end{MyTh}

\begin{MyTh}{LSQ}
  Finding \textit{line} which best fits $\{(a_i,b_i)\}_{i=1}^n$ by \textit{minimise} \comment{$\|\mathbf{Ax}-\mathbf{b}\|_2$}, $A\in\R^{n\times 2}$ vandermonde. \quad
  \textbf{Thm}: \textit{minimises} \ok{$\|\mathbf{Ax}-\mathbf{b}\|_2$ $\Leftrightarrow$} solve \ok{\scriptsize Normal Equations} \ok{$\mathbf{A}^\top\mathbf{Ax}=\mathbf{A}^\top\mathbf{b}$}

  \quad \ok{Moore-Penrose Pseudo-inverse of $A$}: $\mathbf{A}^\dagger = (\mathbf{A}^\top\mathbf{A})^{-1}\mathbf{A}^\top$ \quad $\cdot$ \textit{Normal Equations} has \ok{unique sol} if $\mathbf{A}\mathbf{A}^\top$ invertible. \quad $\cdot$ If $\mathbf{A}$ invertible, \ok{$\mathbf{A}^\dagger=\mathbf{A}^{-1}$}
\end{MyTh}


\subsection{Singular Value Decomposition (SVD)}

\begin{MyTh}{New Def of Condition Number}
  If $\mathbf{A}_{m\times n}$, $m\geq n$ s.t. \ps{it's full rank \imp{\tiny (i.e. $\rank(\mathbf{A})=n$)}}, then \ok{$\kappa_p(\mathbf{A}):=\|\mathbf{A}\|_p\|\mathbf{A}^\dagger\|_p\geq1$} \quad \ok{$\kappa_2(\mathbf{A}^\top\mathbf{A})=\kappa_2(\mathbf{A})^2$} \quad \ok{$\kappa_2(\mathbf{A})=\frac{\sigma_{\max}}{\sigma_{\min}}$}
\end{MyTh}

\vspace{-2pt}
\begin{MyTh}{SVD}
  $\mathbf{A}_{m\times n}=\mathbf{U}\Sigma\mathbf{V}^\top$ \quad where $\mathbf{U}_{m\times m},\mathbf{V}_{n\times n}$ \textit{orthogonal}, $\Sigma_{m\times n}$ \ps{diagonal} with \ps{singular values} $\sigma_1,...,\sigma_p$, $p=\min\{m,n\}$ \quad $\mathbf{U}/\mathbf{V}$的列为 \ps{left/right singular vectors}.

  \quad \footnotesize $^1$ Always exist \quad $^2$ \ok{$\mathbf{A}\mathbf{v}_i=\sigma_i\mathbf{u}_i$} and \ok{$\mathbf{A}^\top\mathbf{u}_i=\sigma_i\mathbf{v}_i$} \quad $^3$ \ok{$\sigma_i^2$|$\mathbf{v}_i$ are eigenvalues/vectors of $\mathbf{A}^\top\mathbf{A}$} \ps{(i.e. $\mathbf{A}^\top\mathbf{A}\mathbf{v}_i=\sigma_i^2\mathbf{v}_i$)} \quad $^4$ $\rank(\mathbf{A})=\rank(\Sigma)=$ \# non-zero $\sigma_i$ \quad $^5$ \ok{$\mathbf{u}_i=\frac{1}{\sigma_i}\mathbf{A}\mathbf{v}_i$}
\end{MyTh}

\begin{MyTh}{Reduced QR Factorisation}
  By QR: $\mathbf{A}_{m\times n}=\mathbf{Q}_{m\times m}\mathbf{R}_{m\times n}$, $m\geq n$. {\scriptsize \quad $^1$ $R$的$n$行以下全0 \quad $^2$ $\mathbf{Q}=[\mathbf{Q}_1 \ \mathbf{Q}_2]$, $\mathbf{R}=[\mathbf{R}_1 \ \mathbf{0}]^{\top}$ $\Rightarrow$ $\mathbf{A}=\mathbf{Q}_1\mathbf{R}_1$ where $\mathbf{Q}_1,\mathbf{R}_1\in \R^{n\times n}$}
\end{MyTh}

\begin{MyTh}{Reduced SVD}
  By SVD: $\mathbf{A}=\mathbf{U}\Sigma\mathbf{V}^\top,m\geq n$. \quad {\scriptsize $^1$ If $\mathbf{B}$ orthogonal, then $\mathbf{B}^{-1}=\mathbf{B}^\top$, $\det(\mathbf{B})=\pm1$ and $\|\mathbf{B}\mathbf{x}\|_2=\|\mathbf{x}\|_2,\forall\mathbf{x}$ \quad $^2$ $\mathbf{U}=[\mathbf{U}_1 \ \mathbf{U}_2]$, $\Sigma=[\Sigma_1 \ \mathbf{0}]^{\top}$ \ $\Rightarrow$ \ $\mathbf{A}=\mathbf{U}_1\Sigma_1\mathbf{V}^\top$, \scalebox{0.8}{\tiny $\mathbf{U}_1\in\R^{m\times n}$, $\Sigma_1\in\R^{n\times n}$}}
\end{MyTh}

\vspace{-1pt}
\begin{MyTh}{Error Analysis of LSQ-QR}
  Let $\widehat{\mathbf{x}}$ denote sol of LSQ-QR (Householder). \quad Then $\widehat{\mathbf{x}}$ minimises $\|(\mathbf{A}+\Delta\mathbf{A})\mathbf{x}-\mathbf{b}\|_2$, and $\frac{\|\Delta\mathbf{A}\|_2}{\|\mathbf{A}\|_2}\leq \alpha n^{\frac{3}{2}}m\varepsilon_m,\alpha>0$
\end{MyTh}

\vspace{-1pt}
\begin{MyTh}{Error Analysis LSQ (any)}
  \tiny Let $\mathbf{x}$|$\widehat{\mathbf{x}}$ minimises $\|\mathbf{Ax}-\mathbf{b}\|_2$|$\|(\mathbf{A}+\Delta\mathbf{A})\mathbf{x}-\mathbf{b}\|_2$ and $\rank(\mathbf{A})=\rank(\mathbf{A}+\Delta\mathbf{A})=n$. \quad If $\kappa_2(\mathbf{A})\frac{\|\Delta\mathbf{A}\|_2}{\|\mathbf{A}\|_2}<1$ \ $\Rightarrow$ \ $\frac{\|\mathbf{x}-\widehat{\mathbf{x}}\|_2}{\|\mathbf{x}\|_2}\leq\frac{\kappa_2(\mathbf{A})}{1-\kappa_2(\mathbf{A})\frac{\|\Delta\mathbf{A}\|_2}{\|\mathbf{A}\|_2}}\cdot\frac{\|\Delta\mathbf{A}\|_2}{\|\mathbf{A}\|_2}\left(2+(\kappa_2(\mathbf{A})+1)\frac{\|\mathbf{r}\|_2}{\|\mathbf{A}\|_2\|\mathbf{x}\|_2}\right)$,$\mathbf{r}=\mathbf{b}-\mathbf{Ax}$
\end{MyTh}
\vspace{-6pt}


\section{Eigenvalue Problems}

\vspace{-8pt}\hspace{120pt}\ps{\tiny (In this section: $|\lambda_1|\ge|\lambda_2|\ge\cdots\ge|\lambda_n|$ are eigenvalues of $\mathbf{A}\in\R^{n\times n}$ with corresponding eigenvectors $\mathbf{x}_1,\mathbf{x}_2,\ldots,\mathbf{x}_n$. $\lambda_1=\lambda_{\max}\leftrightarrow\mathbf{x}_{\max}$, $\lambda_n=\lambda_{\min}\leftrightarrow\mathbf{x}_{\min}$)}

\begin{MyTh}{Rayleigh Quotient}
  $r_{\mathbf{A}}(\mathbf{x})=\frac{\mathbf{x}^\top \mathbf{A} \mathbf{x}}{\mathbf{x}^\top \mathbf{x}}$ \qquad \ok{$r_{\mathbf{A}}(\mathbf{x}_i)=\lambda_i$} \qquad \ps{Eigenvalues $\leftrightarrow$ Eigenvectors}: \ok{$\mathbf{x}_i\to\lambda_i$}: $r_{\mathbf{A}}(\mathbf{x}_i)=\lambda_i$ \quad \ok{$\lambda_i\to\mathbf{x}_i$}: $(\mathbf{A}-\lambda_i\mathbf{I})\mathbf{x}_i=\mathbf{0}$ \quad \ok{Basis}: $\mathbf{A}\mathbf{x}=\sum_i \alpha_i \lambda_i \mathbf{x}_i$
\end{MyTh}

\begin{MyTh}{Convergence of Power Iteration}
  If $|\lambda_1|>|\lambda_2|$ and $\mathbf{x}_1\cdot \mathbf{z}^{(0)}\ne0$, then \ok{$\exists {\sigma_k}_{k\in\N}$} with $\sigma_k=\pm1$ s.t. \ \ok{$\lim\limits_{k\to\infty}\|\sigma_k\mathbf{z}^{(k)}-\mathbf{x}_1\|_2=0$} \ \ and \ \ \ok{$\lim\limits_{k\to\infty}|\lambda^{(k)}-\lambda_1|=0$}
\end{MyTh}

\vspace{-3pt}
\begin{MyTh}{Useful Tool}
  Eigenvalues of $\mathbf{A}^{-1}$ are $\mu_i=\frac{1}{\lambda_i}$ and $\lambda_{\max}=\mu_{\min}^{-1}$ \qquad Eigenvalues of $(\mathbf{A}-s\mathbf{I})^{-1}$ are $\nu_i=\frac{1}{\lambda_i - s}$ and $\lambda_{\max} = s + \nu_{\min}^{-1}$
\end{MyTh}

\vspace{-2pt}
\begin{MyTh}{Inverse Iteration Algorithm}
  Input $\mathbf{A}^{-1}$ in Power Iteration $\Rightarrow$ find $\lambda_{\min}^{-1}$ and $\mathbf{x}_{\min}$ \ps{\tiny (用 SII $s=0$可直接得到)}
\end{MyTh}

\begin{MyTh}{Shifted Inverse Iteration Algorithm}
  Using $(\mathbf{A}-s\mathbf{I})^{-1}$ \ $\Rightarrow$ \ Find eigenvalue \comment{closest to $s$} and corresponding eigenvector. \quad \ps{\footnotesize Assume: $|\lambda_a - s|<|\lambda_b - s|<\cdots$}
\end{MyTh}

\begin{MyTh}{Rate of Convergence for PI \& SII}
  \scriptsize \ok{Rate $\uparrow$ iff $|\frac{\lambda_2}{\lambda_1}|\downarrow$} {\tiny $\|\sigma_k\mathbf{z}^{(k)}-\mathbf{x}_1\|_2\leq\alpha_1(\frac{|\lambda_2|}{|\lambda_1|})^k$ \ ; \ $|\lambda^{(k)}-\lambda_1|\leq\alpha_2(\frac{|\lambda_2|}{|\lambda_1|})^{2k}$} \quad \ok{Rate $\uparrow$ iff $|\frac{\lambda_{a}-s}{\lambda_{b}-s}|\downarrow$} {\tiny $\|\sigma_k\mathbf{z}^{(k)}-\mathbf{x}_i\|_2\leq\beta_1(\frac{|\lambda_{a}-s|}{|\lambda_{b}-s|})^k$ \ ; \ $|\lambda^{(k)}-\lambda_a|\leq\beta_2(\frac{|\lambda_{a}-s|}{|\lambda_{b}-s|})^{2k}$}
\end{MyTh}

\begin{algorithm}[H]
\tiny
\caption*{\textbf{LU} LU factorisation \vspace{-0.1cm} \\ \ps{$C(n) = \sum^{n-1}_{k=1}(n-k)(1+2(n-k+1)) = \frac{2}{3}n^3+\frac{1}{2}n^2-\frac{7}{6}n$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{n \times n}$ \\
\textbf{Output:} $\mathbf{L}, \mathbf{U} \in \mathbb{R}^{n \times n}$ \ \ s.t. \ \ $\mathbf{A} = \mathbf{LU}$
\begin{algorithmic}[1]
  \State $\mathbf{L} = \mathbf{I},\ \mathbf{U} = \mathbf{A}$
  \For{$k = 1, \dots, n - 1$}
    \For{$j = k + 1, \dots, n$}
      \State $l_{jk} = u_{jk} / u_{kk}$
      \State $(u_{jk}, \dots, u_{jn}) = (u_{jk}, \dots, u_{jn}) - l_{jk}(u_{kk}, \dots, u_{kn})$
    \EndFor
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm FS -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{FS} Forward substitution \vspace{-0.1cm} \\ \ps{$C(n) = \sum_{j=1}^{n}[2(j-1)+1]=n^2=\Ocal(n^2)$}}
\textbf{Input:} $\mathbf{L} \in \mathbb{R}^{n \times n}$ \ps{(lower triangular)}, $\mathbf{b} \in \mathbb{R}^n$ \\
\textbf{Output:} $\mathbf{y} \in \mathbb{R}^n$ such that $\mathbf{Ly} = \mathbf{b}$
\begin{algorithmic}[1]
  \For{$j = 1, \dots, n$}
    \State $y_j = \frac{1}{l_{jj}} \left( b_j - \sum_{k = 1}^{j - 1} l_{jk} y_k \right)$
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm BS -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{BS} Back substitution \vspace{-0.1cm} \\ \ps{$C(n) = \sum_{j=1}^n[2(n-j)+1]=n^2=\Ocal(n^2)$}}
\textbf{Input:} $\mathbf{U} \in \mathbb{R}^{n \times n}$ \ps{(upper triangular)}, $\mathbf{y} \in \mathbb{R}^n$ \\
\textbf{Output:} $\mathbf{x} \in \mathbb{R}^n$ such that $\mathbf{Ux} = \mathbf{y}$
\begin{algorithmic}[1]
  \For{$j = n, \dots, 1$}
    \State $x_j = \frac{1}{u_{jj}} \left( y_j - \sum_{k = j + 1}^{n} u_{jk} x_k \right)$
  \EndFor
\end{algorithmic}
\end{algorithm}

% ----- Algorithm (M)GS -----
\begin{algorithm}[H]
\tiny
\caption*{\textbf{(M)GS} (Modified) Gram-Schmidt \quad \ps{$C(n)=2n^3+n^2+n$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{n \times n}$, with columns $\mathbf{a}_1, \dots, \mathbf{a}_n$ \\
\textbf{Output:} $\mathbf{Q} \in \mathbb{R}^{n \times n}$, with orthonormal columns $\mathbf{q}_1, \dots, \mathbf{q}_n$

\qquad\qquad $\mathbf{R} \in \mathbb{R}^{n \times n}$ non-singular upper triangular matrix; \ s.t. $\mathbf{A} = \mathbf{Q}\mathbf{R}$
\begin{algorithmic}[1]
  \State $\mathbf{R} = \mathbf{0}$
  \For{$j=1,...,n$}
    \State $\mathbf{q}_j = \mathbf{a}_j$
    \For{$k=1,...,j-1$}
      \State $r_{kj} = \mathbf{q}_k^\top \mathbf{a}_j$ \quad | \quad ($r_{kj} = \mathbf{q}_k^\top \mathbf{q}_j$)
      \State $\mathbf{q}_j = \mathbf{q}_j - r_{kj} \mathbf{q}_k$ 
    \EndFor
    \State $r_{jj} = \|\mathbf{q}_j\|_2=\sqrt{\mathbf{q}_j^\top \mathbf{q}_j}$
    \State $\mathbf{q}_j = \frac{\mathbf{q}_j}{r_{jj}}$
  \EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithmic}[1]

    \State $\mathbf{v} = \mathbf{u} + \mathrm{sign}(u_1)\|\mathbf{u}\|_2 \mathbf{e}_1$  \quad \textcolor{gray}{\tiny \quad \ \# line.4-7 得到的$\mathbf{Q}_k$:}
    \State $\mathbf{w} = \frac{\mathbf{v}}{\|\mathbf{v}\|_2}$  \qquad\qquad\qquad\quad \textcolor{gray}{\tiny \qquad \# 使得$\mathbf{Q}_k\mathbf{u}=-\text{sign}(u_1)\|\mathbf{u}\|_2\mathbf{e}_1$}

\caption*{\textbf{PI} Power Iteration \quad \ps{$C=?$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{n \times n}$ \textit{symmetric}, $\mathbf{z}^{(0)} \in \mathbb{R}^n\setminus\{\mathbf{0}\}$, $\eps>0$ \\
\textbf{Output:} $\mathbf{z}^{(k)} \in \mathbb{R}^n, \lambda^{(k)} \in \mathbb{R}$ \ s.t. \ \ps{$\mathbf{z}^{(k)} \approx \mathbf{x}_{\max}$, $\lambda^{(k)} \approx \lambda_{\max}$}
\begin{algorithmic}[1]
  \State $k = 0$
  \State $\lambda^{(0)} = r_{\mathbf{A}}(\mathbf{z}^{(0)})$
  \While{$\|\mathbf{A}\mathbf{z}^{(k)} - \lambda^{(k)} \mathbf{z}^{(k)}\|_2 > \eps$} \quad \textcolor{gray}{\tiny \# or: $|\lambda^{(k)} - \lambda^{(k-1)}| > \eps$}
    \State $k = k + 1$ \qquad\qquad\qquad\qquad\qquad \ \ \textcolor{gray}{\tiny \# or: $\|\mathbf{z}^{(k)} - \mathbf{z}^{(k-1)}\|_2 > \eps$}
    \State $\mathbf{z}^{(k)} = \mathbf{A} \mathbf{z}^{(k-1)}$
    \State $\mathbf{z}^{(k)} = \frac{\mathbf{z}^{(k)}}{\|\mathbf{z}^{(k)}\|_2}$ \quad \textcolor{gray}{\tiny \# To improve stability and avoid large values}
    \vspace{-1pt}
    \State $\lambda^{(k)} = r_{\mathbf{A}}(\mathbf{z}^{(k)})$ \qquad \textcolor{red}{\tiny \# In fact: \scalebox{0.8}{$\mathbf{z}^{(k)} =\frac{\mathbf{A}^k \mathbf{z}^{(0)}}{\|\mathbf{A}^k \mathbf{z}^{(0)}\|_2}$ and $\lambda^{(k)} = r_{\mathbf{A}}(\mathbf{z}^{(k)})$}}
    \vspace{-5pt}
  \EndWhile
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\tiny
\caption*{\textbf{SII} Shifted Inverse Iteration \quad \ps{$C=?$}}
\textbf{Input:} $\mathbf{A} \in \mathbb{R}^{n \times n}$ \textit{symmetric}, $s \in \mathbb{R}$, $\mathbf{z}^{(0)} \in \mathbb{R}^n\setminus\{\mathbf{0}\}$, $\eps>0$ \\
\textbf{Output:} $\mathbf{z}^{(k)} \in \mathbb{R}^n, \lambda^{(k)} \in \mathbb{R}$ \ s.t. \ \ps{$\mathbf{z}^{(k)} \approx \mathbf{x}_a$, $\lambda^{(k)} \approx \lambda_a$ near $s$}
\begin{algorithmic}[1]
  \State $k = 0$
  \State $\lambda^{(0)} = r_{\mathbf{A}}(\mathbf{z}^{(0)})$ \quad \textcolor{gray}{\tiny \# first $O(n^3)$ cost; then $O(n^2)$ cost/it}
  \While{$\|\mathbf{A}\mathbf{z}^{(k)} - \lambda^{(k)} \mathbf{z}^{(k)}\|_2 > \eps$}
    \State $k = k + 1$
    \State Solve $(\mathbf{A} - s\mathbf{I}) \mathbf{y} = \mathbf{z}^{(k-1)}$ for $\mathbf{y}$ \quad \textcolor{gray}{\tiny \# via GEPP \& Find LU only once}
    \State $\mathbf{z}^{(k)} = \frac{\mathbf{y}}{\|\mathbf{y}\|_2}$
    \vspace{-1pt}
    \State $\lambda^{(k)} = r_{\mathbf{A}}(\mathbf{z}^{(k)})$ \ \ \textcolor{red}{\tiny \# In fact: \scalebox{0.8}{$\mathbf{z}^{(k)} =\frac{(\mathbf{A}-s\mathbf{I})^{-k} \mathbf{z}^{(0)}}{\|(\mathbf{A}-s\mathbf{I})^{-k} \mathbf{z}^{(0)}\|_2}$, $\lambda^{(k)} = r_{\mathbf{A}}(\mathbf{z}^{(k)})$}}
    \vspace{-5pt}
  \EndWhile
\end{algorithmic}
\end{algorithm}

\begin{MyTh}{Where the error come from calucation}
    \begin{enumerate}
        \item If your calculation involves 15 or more significant digits, you will likely encounter rounding errors. (Since $\varepsilon_m\approx 10^{-16}$ (relative accuracy), which is about 15 significant digits.) 
        \item \textbf{For GE Algorithm}: It is an unstable algorithm. \quad e.g. $\mathbf{A}=[10^{-20} \ \ 1 \ \ \backslash\backslash \ \ 1 \ \ 1] \ \Rightarrow$ By \textbf{GE} Algorithm, $\widehat{\mathbf{L}}\widehat{\mathbf{U}}=\mathbf{A} + [0 \ \ 0 \ \ \backslash\backslash \ \ 0 \ \ -1]$ (error)
        \item \textbf{MGS vs. GS}:In floating point arithmetic, the computed vectors $\{\hat{q}_1, . . . , \hat{q}_{j−1}\}$ will not be orthonormal. Algorithm MGS takes this into account when computing $\hat{q}_j$; Algorithm GS does not.
    \end{enumerate}
\end{MyTh}
