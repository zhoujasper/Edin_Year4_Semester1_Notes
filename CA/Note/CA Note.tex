\section{Basic Knowledge}

\begin{MyTh}{Group $(G,*)$}
    \ok{G1}: Closure \quad \ok{G2}: Associativity \quad \ok{G3}: Identity element \quad \ok{G4}: Inverse element \quad\quad $\downarrow$ \ps{\tiny \ok{Division ring}: $R^{\times}=R\setminus\{0\}$; $R^\times$ group (under $\cdot$)}
\end{MyTh}

\begin{MyTh}{Ring $(R,+,\cdot)$}
    \ok{R1}: $(R,+)$ is an abelian group \quad \ok{R2}: $(R,\cdot)$ is monoid. \ps{(no inverse)} \quad \ok{R3}: Distributive law \ps{(left and right)}
\end{MyTh}

\begin{MyTh}{Ideal $(I)$}
    \ok{I1}: $\ne\emptyset$ \quad \ok{I2}: $\forall a,b\in I, a-b\in I$ \quad \ok{I3}: $\forall r\in R,i\in I, ri,ir\in I$ \qquad
\end{MyTh}
\begin{MyTh}{Lemma}
    \ok{$I=R$ \ $\Leftrightarrow$ \ $1\in I$ \ $\Leftrightarrow$ \ $I\cap R^\times \ne \emptyset$}
\end{MyTh}

{\footnotesize\vspace{-3pt}
\begin{MyTh}{Subgroup $(H)$}
    \ok{H1}: $e\in H$ \quad \ok{H2}: $\forall a,b\in H, ab\in H$ \quad \ok{H3}: $\forall a\in H, a^{-1}\in H$ \qquad
\end{MyTh}
\begin{MyTh}{Subring $(S)$}
    \ok{S1}: $1\in S$ \quad \ok{S2}: $\forall a,b\in S, a-b\in S$ \quad \ok{S3}: $\forall a,b\in S, ab\in S$
\end{MyTh}
}

\begin{MyTh}{Integral Domain (ID)}
    \ok{ID1}: communicative ring \quad \ok{ID2}: no zero divisors \qquad
\end{MyTh}
\begin{MyTh}{Lemma}
    Finite ID $\Rightarrow$ field \quad \comment{ID $\Rightarrow$ Cancellation Law}
\end{MyTh}

\begin{MyTh}{Factor Ring $(R/I)$}
    $a+I=b+I \Leftrightarrow a-b\in I$ \quad $(R/I,\oplus,\odot)$ is a ring. \ps{($1_{R/I}=1_R+I$, $0_{R/I}=0_R+I$, $I$ ideal)} \qquad \imp{$\Z_n\cong \Z/n\Z$}
\end{MyTh}

\begin{MyTh}{Ring Homomorphism}
    $\varphi:R\to S$ s.t. \ok{RH1}: $\varphi(a+b)=\varphi(a)+\varphi(b)$ \quad \ok{RH2}: $\varphi(ab)=\varphi(a)\varphi(b)$ \quad \ok{RH3}: $\varphi(1_R)=1_S$

    \quad $\cdot$ \ok{1}: $\Ker(\varphi)$ is an ideal. \quad \ok{2}: $\Ima(\varphi)$ is a subring. \quad \ok{3}: $\varphi$ is 1-1 $\Leftrightarrow \Ker(\varphi)=\{0_R\}$. \quad \imp{FIT: $R/\Ker(\varphi)\cong \Ima(\varphi)$} \ps{(First Isomorphism Theorem)}

    \quad $\cdot$ \begin{MyTh}{Canonical Homomorphism}
        $can:R\to R/I$ s.t. $can(r)=r+I$ is a onto ring Homomorphism. \quad $\Ker(can)=I$ \qquad $\updownarrow$ \ps{($\exists!\tilde{\varphi}$双射)}
    \end{MyTh}

    \quad $\cdot$ \begin{MyTh}{Lemma}
            If $\varphi:R\to S$ ring Homomorphism s.t. $I\subseteq \Ker(\varphi)$, \ \ then \ \ $\exists! \tilde{\varphi}:R/I\to S$ s.t. $\varphi=\tilde{\varphi}\circ can$ \ \ \comment{(i.e. $\tilde{\varphi}(r+I)=\varphi(r)$)}
    \end{MyTh}
\end{MyTh}


\begin{MyTh}{Lagrange's Theorem}
    $G$ finite group, $H\le G$ subgroup, then $|H| \mid |G|$
\end{MyTh}


\section{Types of Ideals}
\vspace{-13pt}\hspace{90pt} \imp{(Maximal Ideals $\Rightarrow$ Prime Ideals $\Rightarrow$ Radical Ideals $\Rightarrow$ Ideals)}

\begin{MyTh}{Division}
    If $R$ ring, then: \ \ $\forall f,g\in R[x],${\scriptsize $g$首相系数可逆} \ \ $\Rightarrow$ \ \ \ok{$\exists q,r\in R[x]$ s.t. $f(x)=g(x)q(x)+r(x)$ and $\deg(r)<\deg(g)$ or $r=0$.}

    \quad \begin{MyTh}{Lemma}
        If $R$ ring, $g\in R[x],g\ne0$, $I=(g)$. \ $\Rightarrow$ \ $^1$ $\forall i\in R[x]/I,i=f+I,\deg(f)<\deg(g)$
        
        \qquad\qquad\qquad {\scriptsize $g$首相系数可逆} \hspace{104pt} $^2$ If $a,b\in R[x],\deg(a,b)<\deg(g)$ and $a+I=b+I$, then $a=b$. \ps{(系数也相同)}
    \end{MyTh}

    \quad \ok{Remark}: {\tiny If \ps{$I=(g)$, $g$首相系数不可逆}: Find a polynomial $g_1$ s.t. $g_1=hg,h\in R[x]$首相系数可逆, 那么可满足$^1$条件,$^2$只能subset但不能唯一表达.}

    \quad \ok{Remark}: {\scriptsize 有些时候我们能用 $\gcd,\text{lcm}$ 计算 $<a,b>=<\gcd(a,b)>$ 和 $a\cap b=<\text{lcm}(a,b)>$ [If $R$ communicative ring] 但注意! 要在能够用 Division Algorithm | 有 $\gcd,\text{lcm}$ 的环里用.}
\end{MyTh}

\fbox{\parbox[c]{0.98\textwidth}{
\begin{MyTh}{Principal Ideal}
    Ideal $I$: Iff $I=aR$ \quad
\end{MyTh}
\begin{MyTh}{Prime Ideal}
    Ideal $I\ok{\ne R}$: $ab\in I\Rightarrow a\in I$ or $b\in I$. \ \ {\scriptsize (\comment{\scriptsize OR}: If $a\notin I$ and $b\notin I$, then $ab\notin I$)} \quad
\end{MyTh}

\begin{MyTh}{Maximal Ideal}
    Ideal $I\ok{\ne R}$: If $J$ ideal and $I\subseteq J\subseteq R$, then $J=I$ or $J=R$. \qquad\qquad \ps{$\downarrow$ $I$ not radical $\Leftrightarrow$ $\exists a\notin I$ s.t. $a^n\in I$ for some $n\in\N$}
\end{MyTh}

\begin{MyTh}{Radical|Radical Ideal}
    $I\ok{\ne R}$: \ok{Radical}: $\sqrt{I}=\{r\in R:r^n\in I \text{ for some } n\in\N\}$ \quad \ok{Radical Ideal}: If $I=\sqrt{I}$ \qquad $I\subseteq \sqrt{I}$ always.
\end{MyTh}
}}

\begin{MyTh}{Properties}
    Relationship between these kinds of ideals:
    \begin{enumerate}
        \item $R$ communicative ring, $I$ ideal. \ \ \ok{$I$ prime $\Leftrightarrow$ $R/I$ is an ID}. \qquad \ok{$I$ maximal $\Leftrightarrow$ $R/I$ is a field}
        \item {\footnotesize $R$ ring, $I_1,I_2,...,I_n\triangleleft R$, \ok{$I_1\cap I_2\cap\cdots\cap I_n=\{0\}$} \ \ $\Rightarrow$ \ \ $R$ isomorphic to a subring of $R/I_1\oplus R/I_2\oplus\cdots\oplus R/I_n$. \ps{\scriptsize \imp{$R\cong R/I_1\oplus R/I_2\oplus\cdots\oplus R/I_n$} if $I_i+I_j=R$ for $i\ne j$}}
        \item {\small $R$ ring, $I$ ideal, $J$ ideal of $R/I$, $T=\{r\in R:r+I\in J\}$\ps{(reps of $J$)}. \ \ Then $^1$ \ok{$T$: Ideal of $R$} \ \ $^2$ \ok{$I\subseteq T$} \ \ $^3$ \ok{$J=\{t+I:t\in T\}$} \ \ $^4$ \imp{$(R/I)/J\cong R/T$}}
        \qquad {\small If $J=\langle j_1+I,...,j_n+I\rangle$ then $T=\langle j_1,...,j_n\rangle+I=\langle j_1,...,j_n,I\rangle$ \quad Moreover, if $I=\langle i_1,...,i_m\rangle$, then $T=\langle j_1,...,j_n,i_1,...,i_m\rangle$}
        \item $R$ communicative ring, $I$ ideal. \ \ \ok{$I$ radical $\Leftrightarrow$ $R/I$ has no nilpotent elements {\tiny except $0$}} \ \ \ps{\tiny (nilpotent: $r^n=0$)}
        \item If $R$ (commutative with \textit{unity}), then \ok{any ideal $I\ne R$ is contained in a \textit{maximal} ideal.} \qquad \comment{If $r$ is \textit{not a unit}, then $1\notin \langle r\rangle=rR$}
    \end{enumerate}
\end{MyTh}

\section{PID | UFD | Euclidean Domain}
\vspace{-13pt}\hspace{160pt} \imp{(ED $\Rightarrow$ PID $\Rightarrow$ UFD $\Rightarrow$ ID)}

\fbox{\parbox[c]{0.98\textwidth}{
\begin{MyTh}{Irreducible}
    $0\ne a\in R\setminus R^\times$ is \ok{Irreducible}: If $a=bc$, then $b\in R^\times$ or $c\in R^\times$.
\end{MyTh}

\begin{MyTh}{Prime}
    $0\ne a\in R\setminus R^\times$ is \ok{Prime}: If $a\mid bc$, then $a\mid b$ or $a\mid c$. \ \ \ps{\tiny (If $R$ communicative ring, then prime $\Rightarrow$ irreducible)}
\end{MyTh}
}}

\begin{enumerate}
    \item $F$ field, $0\ne g\in F[x]$ non-constant: \ \ $^1$ \ok{$F[x]/\langle g\rangle$ is a field} \ $\Leftrightarrow$ \ $^2$ \ok{$g$ is irreducible in $F[x]$} \ $\Leftrightarrow$ \ $^3$ \ok{$F[x]/\langle g\rangle$ is a domain}
    \item $F$ field, $0\ne g\in F[x]$: \ \ \ok{$g$ is irreducible \ $\Leftrightarrow$ \ $\langle g\rangle$ is prime ideal}
    \item If $R$ is \comment{ID}: \ \ \ok{Prime $\Rightarrow$ Irreducible} \qquad If $0\ne a\in R$, then: \ \ \ok{$a$ prime $\Leftrightarrow$ $aR$ prime ideal}
    \item If $R$ is \comment{PID}, $a\ne 0$: \ \ $^1$ \ok{$aR$ is prime ideal} \ $\Leftrightarrow$ \ $^2$ \ok{$a$ is prime} \ $\Leftrightarrow$ \ $^3$ \ok{$aR$ is maximal ideal} \ \imp{$\Rightarrow$} \ $^4$ \ok{$a$ is irreducible}
\end{enumerate}
\vspace{4pt}

\fbox{\parbox[c]{0.98\textwidth}{
\begin{MyTh}{PID}
    $R$ communicative ring is \textit{principal ideal domain} if: \ \ $\forall I\triangleleft R$, $I=aR$ for some $a\in R$. \qquad Theorem: \ok{$\Z,F,F[x]$ is PID.}
\end{MyTh}

\begin{MyTh}{ED}
    {\small $R$ is \textit{ID}. \ $R$ is \textit{Euclidean Domain} if: \ \ \ok{\scriptsize (Euclidean func)} $\exists v:R\setminus\{0\}\to \N_0$ s.t. $\forall a,b\in R,b\ne0$, $\exists q,r\in R$ s.t. $a=bq+r$ and $r=0$ or $v(r)<v(b)$.}

    \quad \ok{Common ED}: $\Z,F[x]$, $\Z[i]${\tiny Gaussian Integers} are ED, with $v(a)=|a|,v(f)=\deg(f),v(a+bi)=a^2+b^2$ respectively.

    \quad \ok{Euclidean Algorithm}: If $R$ ED, $a,b\in R\setminus\{0\}$, then $\gcd(a,b)=xa+yb$ for some $x,y\in R$. \ps{\tiny (extended of Bezout's Identity)}
\end{MyTh}

\begin{MyTh}{UFD (Def 1)}
    $R$ ID is \textit{unique factorization domain} if: \ok{UFD 1}: $\forall a\in R\setminus\{0\}$, $\exists$ irreducibles $p_1,...,p_n\in R$ s.t. $a=p_1p_2...p_n$

    \quad \ok{UFD 2}: Any other irreducibles $q_1,...,q_m\in R$ s.t. $a=q_1q_2...q_m$, \ \ then \comment{$n=m$ and $p_i=u_iq_i$ for some $u_i\in R^\times$} after reordering.
\end{MyTh}

\begin{MyTh}{Acending Chain Condition (ACC)}
    A sequence of ideals $I_1\subseteq I_2\subseteq I_3\subseteq...$ \textit{satisfies ACC} if: \comment{$\exists n\in\N$ s.t. $I_n=I_{n+1}=I_{n+2}=...$}
\end{MyTh}

\begin{MyTh}{UFD (Def 2)}
    $R$ ID is \textit{UFD} if: \ok{UFD 1}: Every sequence of \textit{Principal Ideals} satisfies \comment{Ascending Chain Condition (ACC)}
    
    \quad \ok{UFD 2}: If $a$ irreducible, \ \ then $a$ is prime \comment{($\Leftrightarrow$ $aR$ is a prime ideal.)} \qquad\qquad\quad \ok{Causs's Theorem: If $R$ is UFD, then $R[x]$ is UFD.}
\end{MyTh}
}}

\begin{enumerate}
    \item \begin{MyTh}{\imp{Bezout's Identity}}
              $R$ is \textit{PID}, then $\forall a,b\in R$, $\exists x,y\in R$ s.t. $ax+by=\gcd(a,b)$.
          \end{MyTh}
    \item $R$ UFD: \ \ \ok{$a$ prime $\Leftrightarrow$ $a$ irreducible} \qquad \qquad $\cdot$ \ok{$R$ PID: Every sequence of ideals satisfies ACC.}
\end{enumerate}

\section{Localisation}

\fbox{\parbox[c]{0.98\textwidth}{
\begin{MyTh}{Field of Fractions ($F_R$)}
    $R$ is ID, $S=R\setminus\{0\}$. \ \ Def \textit{equivalence relation} $\sim$ on $R\times S$ by: \ \ $(a,b)\sim(c,d)$ $\Leftrightarrow$ $ad=bc$. \ \ \comment{\footnotesize (Like $\frac{a}{b}=\frac{c}{d}$)}

    \quad Then the \textit{field of fractions} $F_R$ of $R$: \ \ \ok{$F_R=(R\times S)/\sim \ =\{\frac{a}{b} : a\in R,b\in S\}$} \ \ \comment{where \ $\frac{a}{b}:=\{(a',b'):(a',b')\sim(a,b)\}$}

    \quad \ok{Operations}: \ \ $\frac{a}{b}+\frac{c}{d}=\frac{ad+bc}{bd}$, \ \ $\frac{a}{b}\cdot\frac{c}{d}=\frac{ac}{bd}$, \ \ \comment{Identity}: $0_{F_R}=\frac{0}{1},1_{F_R}=\frac{1}{1}$ \ \ \comment{Inverse}: $-\frac{a}{b}=\frac{-a}{b},(\frac{a}{b})^{-1}=\frac{b}{a}$ if $a\ne0$.

    \quad \ok{Remark}: $\tau:R\to F_R$ by $\tau(r)=\frac{r}{1}$ is a 1-1 ring homomorphism. \quad \ps{ps:Isomorphism to $\{\frac{r}{1}:r\in R\}\subseteq F_R$.} \qquad \ok{$F_R$ is a field.}
\end{MyTh}

\begin{MyTh}{Multiplicative Set}
    $R$ communicative ring. \ \ $X \subseteq R$ is a \textit{multiplicative set} if: \ok{1}: $1_R\in X$ \quad \ok{2}: {\footnotesize 乘法封闭} \quad ( option *\ok{3}: $0_R\notin X$ )
\end{MyTh}

\begin{MyTh}{Ring of Fractions ($X^{-1}R$)}
    $R$ {\tiny 交换环}, $X\subseteq R$ multiplicative set. \ \ Def \textit{e.r.} $\sim$ on $R\times X$ by: \ \ {\footnotesize $(a,x)\sim(b,y)$ $\Leftrightarrow$ $\exists s\in X$ s.t. $s(ay-bx)=0$.}

    \quad Then the \textit{ring of fractions} $X^{-1}R$: \ \ \ok{$X^{-1}R=(R\times X)/\sim \ =\{\frac{a}{x} : a\in R,x\in X\}$} \ \ \comment{where \ $\frac{a}{x}:=\{(a',x'):(a',x')\sim(a,x)\}$}

    \quad \ok{Operations}: \ \ $\frac{a}{x}+\frac{b}{y}=\frac{ay+bx}{xy}$, \ \ $\frac{a}{x}\cdot\frac{b}{y}=\frac{ab}{xy}$, \ \ \comment{Identity}: $0_{X^{-1}R}=\frac{0}{1},1_{X^{-1}R}=\frac{1}{1}$ \ \ \comment{Inverse}: $-\frac{a}{x}=\frac{-a}{x}$ If $a\in X$, then $(\frac{a}{x})^{-1}=\frac{x}{a}$.

    \quad \ok{Remark}: $\varphi:R\to X^{-1}R$ by $\tau(r)=\frac{r}{1}$ is a ring homomorphism. \quad \ps{ps:Isomorphism to $\{\frac{r}{1}:r\in R\}\subseteq X^{-1}R$.} \qquad \ok{$X^{-1}R$ is a ring.}
\end{MyTh}

\begin{MyTh}{Localisation}
    $P$ prime ideal of $A$. \ $\Rightarrow$ \ $^1$\ok{$S=A\setminus P$} is a \textit{multiplicative set}. \ \ $^2$\textit{Localisation of $A$ at $P$}: \ \ \ok{$A_P=S^{-1}A$}
\end{MyTh}
}}

\begin{MyTh}{Theorem}
    $A$ communicative ring, then: \ \ $A_p=S^{-1}A$ has \ok{only one maximal ideal}, which is \ok{$S^{-1}P$}. \ \ \comment{ps: $S=A\setminus P$}
\end{MyTh}


\subsection{Nil and Jacobson Radical Ideals}

\fbox{\parbox[c]{0.98\textwidth}{
\begin{MyTh}{Nil Ideal}
    $I$ ideal of $R$. \ \ $I$ is a \textit{nil ideal} if: \ \ $\forall x\in I$, $\exists n\in\N$ s.t. $x^n=0$. \ \ \comment{i.e. every element of $I$ is nilpotent.} \qquad\quad \ps{$\downarrow$ 用geometric series证明}
\end{MyTh}

\begin{MyTh}{Jacobson Radical Ideal (JRI)}
    $I$ ideal of $R$. \ \ $I$ is a \textit{Jacobson radical ideal} if: \ \ $\forall x\in R$, $1-rx$ is a \textit{unit} in $R$. \qquad \ok{*Nil ideal $\Rightarrow$ JRI}
\end{MyTh}
}}

\begin{MyTh}{Zorn's Lemma}
     If a \textit{partially ordered set} \ps{\tiny (i.e. $\exists\sim_{\leq}$)} $P$ s.t. \ \ $^1$ $\forall$ \textit{chain} in $P$ has \textit{upper bound}. \ \ $^2$ $P\ne\emptyset$. \ $\Rightarrow$ \ $P$ has at least one \textit{maximal element}. \ps{(i.e. $\exists m\in P$ s.t. $\forall p\in P$, if $m\leq p$, then $m=p$)}
\end{MyTh}
