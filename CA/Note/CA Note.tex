\begin{MyTh}{Ring Homomorphism}

    \end{MyTh}
\end{MyTh}

\begin{MyTh}{Division}
    If $R$ ring, then: \ \ $\forall f,g\in R[x],${\scriptsize $g$首相系数可逆} \ \ $\Rightarrow$ \ \ \ok{$\exists q,r\in R[x]$ s.t. $f(x)=g(x)q(x)+r(x)$ and $\deg(r)<\deg(g)$ or $r=0$.}

    \quad \begin{MyTh}{Lemma}
        If $R$ ring, $g\in R[x],g\ne0$, $I=(g)$. \ $\Rightarrow$ \ $^1$ $\forall i\in R[x]/I,i=f+I,\deg(f)<\deg(g)$
        
        \qquad\qquad\qquad {\scriptsize $g$首相系数可逆} \hspace{104pt} $^2$ If $a,b\in R[x],\deg(a,b)<\deg(g)$ and $a+I=b+I$, then $a=b$. \ps{(系数也相同)}
    \end{MyTh}

    \quad \ok{Remark}: {\tiny If \ps{$I=(g)$, $g$首相系数不可逆}: Find a polynomial $g_1$ s.t. $g_1=hg,h\in R[x]$首相系数可逆, 那么可满足$^1$条件,$^2$只能subset但不能唯一表达.}

    \quad \ok{Remark}: {\scriptsize 有些时候我们能用 $\gcd,\text{lcm}$ 计算 \ok{$\langle a,b\rangle=\langle\gcd(a,b)\rangle$} 和 \ok{$\langle a\rangle\cap \langle b\rangle=\langle\text{lcm}(a,b)\rangle$} [If $R$ communicative ring] 但注意! 要在能够用 Division Algorithm | 有 $\gcd,\text{lcm}$ 的环里用.}
\end{MyTh}

\begin{MyTh}{Prime Ideal}

\end{MyTh}

\begin{MyTh}{Maximal Ideal}

\end{MyTh}

\begin{MyTh}{Radical|Radical Ideal}

\end{MyTh}
}}

\begin{MyTh}{Properties}
    Relationship between these kinds of ideals:
    \begin{enumerate}
        \item $R$ communicative ring, $I$ ideal. \ \ \ok{$I$ prime $\Leftrightarrow$ $R/I$ is an ID}. \qquad \ok{$I$ maximal $\Leftrightarrow$ $R/I$ is a field}
        \item {\scriptsize $R$ ring, $I_1,I_2,...,I_n\triangleleft R$, and \ok{$I_1\cap I_2\cap\cdots\cap I_n=I\triangleleft R$} \ \ $\Rightarrow$ \ \ $R\cong$ \textit{subring of} $R/I_1\oplus R/I_2\oplus\cdots\oplus R/I_n$. \ | Moreover, \ \ps{\scriptsize \imp{$R/I\cong R/I_1\oplus R/I_2\oplus\cdots\oplus R/I_n$} if $I_i+I_j=R$ for $i\ne j$}}
        \item {\small $R$ ring, $I$ ideal, $J$ ideal of $R/I$, $T=\{r\in R:r+I\in J\}$\ps{(reps of $J$)}. \ \ Then $^1$ \ok{$T$: Ideal of $R$} \ \ $^2$ \ok{$I\subseteq T$} \ \ $^3$ \ok{$J=\{t+I:t\in T\}$} \ \ $^4$ \imp{$(R/I)/J\cong R/T$}}
        \qquad {\small If $J=\langle j_1+I,...,j_n+I\rangle$ then $T=\langle j_1,...,j_n\rangle+I=\langle j_1,...,j_n,I\rangle$ \quad Moreover, if $I=\langle i_1,...,i_m\rangle$, then $T=\langle j_1,...,j_n,i_1,...,i_m\rangle$}
        \item $R$ commutative ring, $I$ ideal. \ \ \ok{$I$ radical $\Leftrightarrow$ $R/I$ has no nonzero nilpotent elements {\tiny except $0$}} \ \ \ps{\tiny (nilpotent: $r^n=0$)}
        \item If $R$ (commutative with \textit{unity}), then \ok{any ideal $I\ne R$ is contained in a \textit{maximal} ideal.} \qquad \comment{If $r$ is \textit{not a unit}, then $1\notin \langle r\rangle=rR$}
    \end{enumerate}
\end{MyTh}

\section{PID | UFD | Euclidean Domain}
\vspace{-13pt}\hspace{160pt} \imp{(ED $\Rightarrow$ PID $\Rightarrow$ UFD $\Rightarrow$ ID)}

\fbox{\parbox[c]{0.98\textwidth}{
\begin{MyTh}{Irreducible}
    $0\ne a\in R\setminus R^\times$ is \ok{Irreducible}: If $a=bc$, then $b\in R^\times$ or $c\in R^\times$.
\end{MyTh}

\begin{MyTh}{Prime}
    $0\ne a\in R\setminus R^\times$ is \ok{Prime}: If $a\mid bc$, then $a\mid b$ or $a\mid c$. \ \ \ps{\tiny (If $R$ communicative ring, then prime $\Rightarrow$ irreducible)}
\end{MyTh}
}}

\begin{enumerate}
    \item $F$ field, $0\ne g\in F[x]$ non-constant: \ \ $^1$ \ok{$F[x]/\langle g\rangle$ is a field} \ $\Leftrightarrow$ \ $^2$ \ok{$g$ is irreducible in $F[x]$} \ $\Leftrightarrow$ \ $^3$ \ok{$F[x]/\langle g\rangle$ is a domain}
    \item $F$ field, $0\ne g\in F[x]$: \ \ \ok{$g$ is irreducible \ $\Leftrightarrow$ \ $\langle g\rangle$ is prime ideal}
    \item If $R$ is \comment{ID}: \ \ \ok{Prime $\Rightarrow$ Irreducible} \qquad If $0\ne a\in R$, then: \ \ \ok{$a$ prime $\Leftrightarrow$ $aR$ prime ideal}
    \item If $R$ is \comment{PID}, \imp{$a\ne 0$}: \ \ $^1$ \ok{$aR$ is prime ideal} \ $\Leftrightarrow$ \ $^2$ \ok{$a$ is prime} \ $\Leftrightarrow$ \ $^3$ \ok{$aR$ is maximal ideal} \ $\Leftrightarrow$ \ $^4$ \ok{$a$ is irreducible} \quad \imp{\tiny 要单独考虑$I=\{0\}$}
\end{enumerate}
\vspace{4pt}

\fbox{\parbox[c]{0.98\textwidth}{
\begin{MyTh}{PID}
    $R$ commutative ring is \textit{principal ideal domain} if: \ \ $\forall I\triangleleft R$, $I=aR$ for some $a\in R$. \qquad Theorem: \ok{$\Z,F,F[x]$ is PID.}
\end{MyTh}

\begin{MyTh}{ED}
    {\small $R$ is \textit{ID}. \ $R$ is \textit{Euclidean Domain} if: \ \ \ok{\scriptsize (Euclidean func)} $\exists v:R\setminus\{0\}\to \N_0$ s.t. $\forall a,b\in R,b\ne0$, $\exists q,r\in R$ s.t. $a=bq+r$ and $r=0$ or $v(r)<v(b)$.}

    \quad \ok{Common ED}: $\Z,F[x]$, $\Z[i]${\tiny Gaussian Integers} are ED, with $v(a)=|a|,v(f)=\deg(f),v(a+bi)=a^2+b^2$ respectively.

    \quad \ok{Euclidean Algorithm}: If $R$ ED, $a,b\in R\setminus\{0\}$, then $\gcd(a,b)=xa+yb$ for some $x,y\in R$. \ps{\tiny (extended of Bezout's Identity)}
\end{MyTh}

\begin{MyTh}{UFD (Def 1)}
    $R$ ID is \textit{unique factorization domain} if: \ok{UFD 1}: $\forall a\in R\setminus\{0\}$, $\exists$ irreducibles $p_1,...,p_n\in R$ s.t. $a=p_1p_2...p_n$

    \quad \ok{UFD 2}: Any other irreducibles $q_1,...,q_m\in R$ s.t. $a=q_1q_2...q_m$, \ \ then \comment{$n=m$ and $p_i=u_iq_i$ for some $u_i\in R^\times$} after reordering.
\end{MyTh}

\begin{MyTh}{Acending Chain Condition (ACC)}
    A sequence of ideals $I_1\subseteq I_2\subseteq I_3\subseteq...$ \textit{satisfies ACC} if: \comment{$\exists n\in\N$ s.t. $I_n=I_{n+1}=I_{n+2}=...$}
\end{MyTh}

\begin{MyTh}{UFD (Def 2)}
    $R$ ID is \textit{UFD} if: \ok{UFD 1}: Every sequence of \textit{Principal Ideals} satisfies \comment{Ascending Chain Condition (ACC)}
    
    \quad \ok{UFD 2}: \imp{If $a$ irreducible, \ \ then $a$ is prime} \comment{($\Leftrightarrow$ $aR$ is a prime ideal.)} \qquad\qquad \ \ \ok{Causs's Theorem: If $R$ is UFD, then $R[x]$ is UFD.}
\end{MyTh}
}}

\begin{enumerate}
    \item \begin{MyTh}{\imp{Bezout's Identity}}
              $R$ is \textit{PID}, then $\forall a,b\in R$, $\exists x,y\in R$ s.t. $ax+by=\gcd(a,b)$. \ \scalebox{0.9}{\ps{\tiny In $\F[...,x_n]: \text{lcm}(f,g)=\frac{fg}{\gcd(f,g)}, \langle f,g\rangle=\langle \text{gcd}(f,g)\rangle,\langle f\rangle \cap \langle g\rangle=\langle \text{lcm}(f,g)\rangle$}}
          \end{MyTh}
    \item $R$ UFD: \ \ \ok{$a$ prime $\Leftrightarrow$ $a$ irreducible} \qquad \qquad $\cdot$ \ok{$R$ PID: Every sequence of ideals satisfies ACC.}
\end{enumerate}

\section{Localisation}

\fbox{\parbox[c]{0.98\textwidth}{
\begin{MyTh}{Field of Fractions ($F_R$)}
    $R$ is ID, $S=R\setminus\{0\}$. \ \ Def \textit{equivalence relation} $\sim$ on $R\times S$ by: \ \ $(a,b)\sim(c,d)$ $\Leftrightarrow$ $ad=bc$. \ \ \comment{\footnotesize (Like $\frac{a}{b}=\frac{c}{d}$)}

    \quad Then the \textit{field of fractions} $F_R$ of $R$: \ \ \ok{$F_R=(R\times S)/\sim \ =\{\frac{a}{b} : a\in R,b\in S\}$} \ \ \comment{where \ $\frac{a}{b}:=\{(a',b'):(a',b')\sim(a,b)\}$}

    \quad \ok{Operations}: \ \ $\frac{a}{b}+\frac{c}{d}=\frac{ad+bc}{bd}$, \ \ $\frac{a}{b}\cdot\frac{c}{d}=\frac{ac}{bd}$, \ \ \comment{Identity}: $0_{F_R}=\frac{0}{1},1_{F_R}=\frac{1}{1}$ \ \ \comment{Inverse}: $-\frac{a}{b}=\frac{-a}{b},(\frac{a}{b})^{-1}=\frac{b}{a}$ if $a\ne0$.

    \quad \ok{Remark}: $\tau:R\to F_R$ by $\tau(r)=\frac{r}{1}$ is a 1-1 ring homomorphism. \quad \ps{ps:Isomorphism to $\{\frac{r}{1}:r\in R\}\subseteq F_R$.} \qquad \ok{$F_R$ is a field.}
\end{MyTh}

\begin{MyTh}{Multiplicative Set}
    $R$ commutative ring. \ \ $X \subseteq R$ is a \textit{multiplicative set} if: \ok{1}: $1_R\in X$ \quad \ok{2}: {\footnotesize 乘法封闭} \quad ( option *\ok{3}: $0_R\notin X$ )
\end{MyTh}

\begin{MyTh}{Ring of Fractions ($X^{-1}R$)}
    $R$ {\tiny 交换环}, $X\subseteq R$ multiplicative set. \ \ Def \textit{e.r.} $\sim$ on $R\times X$ by: \ \ {\footnotesize $(a,x)\sim(b,y)$ $\Leftrightarrow$ $\exists s\in X$ s.t. $s(ay-bx)=0$.}

    \quad Then the \textit{ring of fractions} $X^{-1}R$: \ \ \ok{$X^{-1}R=(R\times X)/\sim \ =\{\frac{a}{x} : a\in R,x\in X\}$} \ \ \comment{where \ $\frac{a}{x}:=\{(a',x'):(a',x')\sim(a,x)\}$}

    \quad \ok{Operations}: \ \ $\frac{a}{x}+\frac{b}{y}=\frac{ay+bx}{xy}$, \ \ $\frac{a}{x}\cdot\frac{b}{y}=\frac{ab}{xy}$, \ \ \comment{Identity}: $0_{X^{-1}R}=\frac{0}{1},1_{X^{-1}R}=\frac{1}{1}$ \ \ \comment{Inverse}: $-\frac{a}{x}=\frac{-a}{x}$ If $a\in X$, then $(\frac{a}{x})^{-1}=\frac{x}{a}$.

    \quad \ok{Remark}: $\varphi:R\to X^{-1}R$ by $\tau(r)=\frac{r}{1}$ is a ring homomorphism. \quad \ps{ps:Isomorphism to $\{\frac{r}{1}:r\in R\}\subseteq X^{-1}R$.} \qquad \ok{$X^{-1}R$ is a ring.}
\end{MyTh}

\begin{MyTh}{Localisation}
    $P$ prime ideal of $A$. \ $\Rightarrow$ \ $^1$\ok{$S=A\setminus P$} is a \textit{multiplicative set}. \ \ $^2$\textit{Localisation of $A$ at $P$}: \ \ \ok{$A_P=S^{-1}A$}
\end{MyTh}
}}

\begin{MyTh}{Theorem}
    $A$ communicative ring, then: \ \ $A_p=S^{-1}A$ has \ok{only one maximal ideal}, which is \ok{$S^{-1}P$}. \ \ \comment{ps: $S=A\setminus P$}
\end{MyTh}

\subsection{Nil and Jacobson Radical Ideals, Local Ring}

\fbox{\parbox[c]{0.98\textwidth}{

\begin{MyTh}{Nil Ideal}
    $I$ ideal of $R$ is a \textit{nil ideal} if: \ \ $\forall x\in I$, $\exists n\in\N$ s.t. $x^n=0$. \ \ {\scriptsize i.e. every element of $I$ is nilpotent.}
\end{MyTh}

\begin{MyTh}{Nil Radical ($N(R)=\sqrt{0_R}$)}
    $N(R):=$ the \comment{largest nil ideal} in $R$ \ \ $\Leftrightarrow$ \ \ $N(R):=$ \comment{sum of all nil ideals in $R$} \ \ $\Leftrightarrow$ \ \ $N(R)=\sqrt{0_R}$\

    \quad \textbf{Note}: \comment{sum of any two nil ideals is also a nil ideal.} \qquad \textbf{Note}: \comment{$R/N(R)$ has no non-zero nilpotent elements.}
\end{MyTh}

\begin{MyTh}{Quasi-invertible}
    $R$ ring, $a\in R$ is \textit{quasi-invertible} if: \ \ $\exists b\in R$ s.t. $(1-a)(1-b)=1$ \ \ $\leftrightarrow$ \ \ $ab=a+b$. \qquad \ps{\tiny (如果a quasi-invertible, 那么b唯一)}
\end{MyTh}

\begin{MyTh}{Jacobson Radical Ideal}
    $I$ ideal of $R$ is a \textit{Jacobson radical ideal} if: \ \ $\forall i\in I$, $i$ is \textit{quasi-invertible} in $R$.
\end{MyTh}

\begin{MyTh}{Jacobson Radical $J(R)$}
    $J(R):=$ the \comment{largest JRI} in $R$ \ \ $\Leftrightarrow$ \ \ $J(R):=$ \comment{sum of all JRI in $R$} \ \ $\Leftrightarrow$ \ \ \ok{$J(R)=\bigcap$ {\tiny (left)} maximal ideals of $R$.}

    \quad \textbf{Note}: \comment{sum of any two JRI is also a JRI.} \qquad \textbf{Note}: \ok{Nil ideal $\Rightarrow$ Jacobson Radical Ideal (JRI).}\ps{\tiny (通过等比数列证明)}
\end{MyTh}

\begin{MyTh}{Local Ring}
    $R$ is a \textit{local ring} if: \ \ $R$ has \ok{unique maximal ideal} $\Leftrightarrow$ $R/J(R)$ is a field \qquad {\footnotesize \ok{$A_P$ is a local ring.}\ps{\tiny ($A$ commutative ring, $P$ prime ideal of $A$)}}
\end{MyTh}
}}

\textbf{Remark}: $R/(JRI)$ or $R/J(R)$ has \comment{no non-zero nilpotent elements} \qquad \ok{$(1-a)(1-b)=1 \Leftrightarrow$ $1-a$ is invertible} \ps{($b=1-(1-a)^{-1}$)}


\subsection{Variety \& Vanishing Ideal}

\begin{MyTh}{Variety}
    Let $k$ be a field, $f_1,...,f_m\in k[x_1,...,x_n]$. Then: \ \ \ok{Variety}: $\mathbf{V}(f_1,...,f_m):=\{(a_1,...,a_n)\in k^n:f_i(a_1,...,a_n)=0,\forall i\}$
\end{MyTh}

\begin{MyTh}{Vanishing Ideal}
    Let $k$ be a field, $X\subseteq k^n$. Then: \ \ \ok{Vanishing Ideal}: $\specialI(X):=\{f\in k[x_1,...,x_n]:$ {\footnotesize $f(a_1,...,a_n)=0,\forall (a_1,...,a_n)\in X$ }$\}$
\end{MyTh}
}}

\begin{MyTh}{Useful Tools}
    If \ok{$X=X_1\cup X_2\cup\cdots\cup X_m$}, then \ \ \ok{$\specialI(X)=\specialI(X_1)\cap \specialI(X_2)\cap \cdots \cap \specialI(X_m)$} \qquad $\cdot$ If $I_1\subseteq I_2$ ideals, then \ \ \ok{$\mathbf{V}(I_2)\subseteq \mathbf{V}(I_1)$}

    \hspace{203pt} \ok{$V(I_1\cap I_2 \cap \cdots \cap I_m)=V(I_1)\cup V(I_2)\cup \cdots \cup V(I_m)$}
\end{MyTh}


\subsection{Nullstellensatz}

\begin{MyTh}{Weak Nullstellensatz}
    $\F$ field, and $I\triangleleft \F[x_1,...,x_n]$ ideal. Then: \ok{$(\Rightarrow)$} If $I=\langle x_1-a_1,...,x_n-a_n\rangle$, $a_i\in \F$, then $I$ is maximal ideal.

    \quad \ok{Weak Nullstellensatz} \ok{$(\Leftarrow)$} If $I$ is \textit{maximal ideal} and \underline{$\F$ is algebraically closed}, then \ok{$I=\langle x_1-a_1,...,x_n-a_n\rangle$} for some $a_i\in \F$.

    \quad \ok{Fact I}: If $I=\langle x_1-a_1,...,x_n-a_n\rangle$, then \ \ \ok{[ $f(a_1,...,a_n)=0$ \ \ $\Leftrightarrow$ \ \ $f\in I$ ]}

    \quad \ok{Fact II}: $k$ field, \ok{$(\Rightarrow)$} $I$ \comment{maximal ideal} then \ \ \ok{$\mathbf{V}(I)$ is a \textit{single point}}. \ \ \ps{(i.e. $\mathbf{V}(I)=\{(a_1,...,a_n)\}$)} \qquad \ok{\footnotesize $(\Leftarrow)$ if $k$ algebraically closed.} \ps{\tiny (ps:最好证明一下)}
\end{MyTh}

\begin{MyTh}{Reformation of Weak Nullstellensatz}
    $k$ \textit{algebraically closed} field, $J$ ideals of $k[x_1,...,x_n]$. Then: 
    
    \quad $^1$ \ok{$\mathbf{V}(J)\ne\emptyset$ $\Leftrightarrow$ $J\ne k[x_1,...,x_n]$} \qquad\qquad $^2$ \ok{$f\in \specialI(\mathbf{V}(J))$ \ \ $\Leftrightarrow$ \ \ $f^n\in J$ for some $n\in\N$} \quad i.e. \imp{$\sqrt{J}=\specialI(\mathbf{V}(J))$} \quad \ps{It's \underline{Strong Nullstellensatz}!}
\end{MyTh}

\begin{MyTh}{Reformation of Hilbert's Nullstellensatz}
    $k$ \textit{algebraically closed} field, $I$ ideal of $k[x_1,...,x_n]$. Then:

    \quad $^1$ \ok{$\sqrt{I}=\bigcap$ (maximal ideals $\supseteq I$)} \qquad $\Leftrightarrow$ \qquad $^2$ \ok{$J(R/I)=\bigcap$ maximal ideals in $R/I$ = nil radical of $R/I = \sqrt{0_{R/I}}$}
\end{MyTh}

\begin{MyTh}{2nd Def of Radical}
    $R$ commutative ring, $I$ ideal of $R$. \ \ Then: \ \ \ok{$\sqrt{I}=\bigcap$ (prime ideals $\supseteq I$)}\quad

    \quad
    \begin{MyTh}{Corollary}
        If $R$ commutative ring, then \ok{Nil radical $\sqrt{0_R}=N(R)=\bigcap$ prime ideals of $R$ $=:$ Prime Radical}. \ \ \ps{\tiny If $R$ 非交换,不一定.}
    \end{MyTh}
\end{MyTh}

\begin{MyTh}{Monomials}

\begin{MyTh}{Gröbner Basis}

    \quad \ok{Reduced Gröbner Basis}: If G is Gröbner basis of I and no \textit{proper subset} of G is Gröbner basis of I, then G is a \textit{reduced} GB of $I$.

    \qquad \ok{Remark}: $G$ is Gröbner basis of $I$ if: $^1$ $G$ generates $I$; $^2$ S-polynomial of any pair in $G$ reduces to $0$ by $G$.
\end{MyTh}

\begin{MyTh}{Algorithm to Find Gröbner Basis}
    For ideal $I=\langle f_1,f_2,...,f_m\rangle \triangleleft k[x_1,...,x_n]$, $k$ field. \ \ e.g. \ps{Buchberger's Algorithm}
    \begin{enumerate}
        \item Let $G=\{f_1,f_2,...,f_m\}$
        \item For each $g_i\in G$, write $g_i=LC(g_i) \cdot LM(g_i)+h_i$ \hspace{150pt} \ps{ps: $Ov(g_i,g_j)=Ov(LM(g_i),LM(g_j))$}
        \item For each pair $(g_i,g_j)$ in $G$, calculate $S(g_i,g_j)$ \ps{only $i>j$ or $i<j$ to avoid repetition} \quad \ok{Remark}: If $Ov(g_i,g_j)=1$, then $S(g_i,g_j)\to0$, skip it.
        \item Reduce each $S(g_i,g_j)$: \ \  If $LM(g_k)$ divides any term of $S(g_i,g_j)$, \comment{substitute $LM(g_k)$ by $-h_k$}; until no $LM(g_k)$ divides any term.
        \item If $S(g_i,g_j)\ne0$ after reduction, add it's \comment{leading monomial} to $G$ and go to \comment{step 2}. \qquad If all $S(g_i,g_j)=0$, stop and output $G$.
    \end{enumerate}
\end{MyTh}

\begin{MyTh}{Brönner Basis Application}
    $k$ field, $I\triangleleft k[x_1,...,x_n]$, $G$ Gröbner basis of $I$. \ \ $L$ \comment{leading monomial} of $G$. \ \ $W$ set of \comment{monomials not divisible by any element of $L$}. \ps{(i.e. don't contain any monomials in $L$ as a \textit{subword})} \quad \ps{ps:$1$ is monomial}

\end{MyTh}

\begin{MyTh}{Well-Defined Ordering}
    A \textit{well-defined} ordering $<$ on set $S$ if: \ \ \ok{O1}: $\forall a,b\in S$, $a\ne b$, then $a<b$ or $b<a$ \quad \ok{O2}: $\forall a,b,c\in S$, if $a<b$ and $b<c$, then $a<c$ \quad \ok{O3}: (For Grobner Basis) if $a<b$, then $ac<bc$ for any $c\in S$. \quad \ok{O4}: Every strictly decreasing sequence $a_1>a_2>a_3>...$ in $S$ is finite. (monomialsterminates)
\end{MyTh}
